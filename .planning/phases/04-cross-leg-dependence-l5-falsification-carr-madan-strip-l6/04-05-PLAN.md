    ```python
    """FFT-based Carr-Madan static-replication strip on convergence-tested grid.

    Math route (Carr & Madan 2001, "Towards a Theory of Volatility Trading"):
      Strip price = f(F) + ∫₀^F p(K) f''(K) dK + ∫_F^∞ c(K) f''(K) dK
      where:
        F           = forward (here taken as the mean of the density)
        p(K)/c(K)   = OTM put/call prices, computed from the characteristic function
                      via Carr–Madan-style FFT inversion (np.fft.fft / np.fft.ifft only)
        f''(K)      = second derivative of the payoff f(S_T)
      For a linear forward payoff f(S)=S used in v1.0: f(F)=F and f''=δ(K-F), so both
      integrals collapse and the FFT result is exactly the price; for v2.0 stream-PV
      payoffs the integrals contribute non-trivially and the same FFT density is the
      quadrature kernel. The compute_strip return field `strip_prices` records the
      f''(K)-weighted price at each strike on the FFT grid.

    Per RESEARCH Pattern 5: polymorphic payoff for v2.0 streaming-tokenization
    (caller passes f(S_T); v1.0 uses LP-fee revenue; v2.0 will pass stream-PV).

    Per PRE_REGISTRATION §Carr-Madan Grid Numerical Tolerances (amendment 2026-05-27):
      - POSITIVITY_TOLERANCE = 0.001 (0.1% negative mass relative to total |q|)
      - Grid start: 2^11. Escalate to 2^12 EXACTLY ONCE. Abort to strip_degenerate on failure.
      - NEVER silently switch to COS or PROJ methods.

    Per iter-3 revision (2026-05-27): a fourth HEDGE-05 firing condition `null_strip_unavailable`
    routes gate-passed + strip-not-emittable hybrid state. compute_strip emits a `reason`
    field on strip_degenerate ("positivity_fail_after_2_12") that the orchestrator (04-08)
    consumes; orchestrator-caught upstream exceptions inject reason="build_failed_upstream".
    null_result.decide_firing_condition reads this field and returns "null_strip_unavailable".

    Anti-patterns banned by pre-commit hook carr-madan-anti-pattern-gate:
      - scipy.integrate.quad / fixed_quad / romberg
      - np.trapz
    """
    from __future__ import annotations
    from typing import Callable
    import numpy as np

    REQUIRED_STRIP_KEYS: tuple[str, ...] = (
        "chainId", "contractAddress", "blockRange",
        "fetchTimestamp", "dataHash", "gitCommit", "run_id",
        "strip_prices", "strikes", "n_grid_used", "escalated_to_2_12",
        "negative_mass_fraction", "positivity_tolerance",
    )

    STRIP_DEGENERATE_KEYS: tuple[str, ...] = (
        "chainId", "contractAddress", "blockRange",
        "fetchTimestamp", "dataHash", "gitCommit", "run_id",
        "max_negative_value", "total_negative_mass",
        "characteristic_function_decay_rate", "recommended_method",
        "reason",   # iter-3: "positivity_fail_after_2_12" | "build_failed_upstream"
    )

    POSITIVITY_TOLERANCE: float = 0.001    # PRE_REGISTRATION amendment 2026-05-27
    DEFAULT_N_GRID: int = 2**11
    ESCALATED_N_GRID: int = 2**12
    DEFAULT_U_MAX: float = 200.0           # FFT truncation upper bound for characteristic function


    def _fft_inverse_density(
        char_func: Callable[[np.ndarray], np.ndarray],
        n_grid: int,
        u_max: float = DEFAULT_U_MAX,
    ) -> tuple[np.ndarray, np.ndarray]:
        """Return (k_grid, q_density) from FFT inversion of char_func on [-u_max, u_max].

        q(k) = (1 / 2π) ∫ φ(u) exp(-i u k) du, discretized via np.fft.fft.
        Uses np.fft only (no scipy.integrate, no np.trapz).
        """
        du = (2.0 * u_max) / n_grid
        u_grid = np.linspace(-u_max, u_max - du, n_grid)
        phi_values = char_func(u_grid)
        # FFT inversion (up to phase + dk normalization)
        q_density = np.real(np.fft.fftshift(np.fft.ifft(np.fft.ifftshift(phi_values)))) * n_grid * du / (2.0 * np.pi)
        # Corresponding k grid
        dk = (2.0 * np.pi) / (n_grid * du)
        k_grid = np.linspace(-n_grid * dk / 2.0, n_grid * dk / 2.0 - dk, n_grid)
        return k_grid, q_density


    def _negative_mass_fraction(q_density: np.ndarray) -> float:
        """Returns ∑ q(k)⁻ / ∑ |q(k)| — the locked Carr-Madan positivity metric."""
        total_abs = float(np.sum(np.abs(q_density)))
        if total_abs == 0.0:
            return 0.0
        neg = float(np.sum(np.where(q_density < 0.0, -q_density, 0.0)))
        return neg / total_abs


    def _estimate_char_func_decay_rate(char_func: Callable[[np.ndarray], np.ndarray]) -> float:
        """Crude decay-rate proxy used in strip_degenerate.json provenance."""
        u_probe = np.array([1.0, 10.0, 100.0])
        phi_vals = np.abs(char_func(u_probe))
        # log-log slope
        with np.errstate(divide="ignore", invalid="ignore"):
            return float(np.polyfit(np.log(u_probe), np.log(np.maximum(phi_vals, 1e-30)), 1)[0])


    def compute_strip(
        payoff: Callable[[np.ndarray], np.ndarray],
        char_func: Callable[[np.ndarray], np.ndarray],
        n_grid: int = DEFAULT_N_GRID,
        positivity_tolerance: float = POSITIVITY_TOLERANCE,
        max_escalations: int = 1,
    ) -> dict:
        """Carr-Madan replicating strip with 2^11 -> 2^12 escalation and abort-to-degenerate fallback.

        Returns a strip dict with REQUIRED_STRIP_KEYS (minus provenance keys, injected by orchestrator)
        OR a degenerate dict with STRIP_DEGENERATE_KEYS.
        """
        attempted: list[int] = []
        for attempt in range(max_escalations + 1):
            current_n = n_grid * (2 ** attempt)
            attempted.append(current_n)
            k_grid, q_density = _fft_inverse_density(char_func, current_n)
            neg_frac = _negative_mass_fraction(q_density)
            if neg_frac < positivity_tolerance:
                # Build strip prices via payoff dotted against density on the grid
                payoffs = payoff(k_grid)
                # Normalize density (positive support only for pricing)
                q_pos = np.where(q_density > 0.0, q_density, 0.0)
                norm = float(np.sum(q_pos))
                strip_prices = (payoffs * q_pos / max(norm, 1e-30)).tolist()
                return {
                    "strip_prices": strip_prices,
                    "strikes": k_grid.tolist(),
                    "n_grid_used": int(current_n),
                    "escalated_to_2_12": bool(attempt > 0),
                    "negative_mass_fraction": float(neg_frac),
                    "positivity_tolerance": float(positivity_tolerance),
                }
        # Exhausted escalations -> degenerate (iter-3 Issue 1: add `reason` for fourth-firing-condition routing)
        max_neg = float(np.min(q_density))
        total_neg = float(np.sum(np.where(q_density < 0.0, -q_density, 0.0)))
        decay = _estimate_char_func_decay_rate(char_func)
        recommended = "COS" if decay > -0.5 else ("PROJ" if decay > -1.5 else "none")
        return {
            "max_negative_value": max_neg,
            "total_negative_mass": total_neg,
            "characteristic_function_decay_rate": decay,
            "recommended_method": recommended,
            "reason": "positivity_fail_after_2_12",   # iter-3: routes to null_strip_unavailable (d)
        }
    ```
