    ```python
    """Four-condition convex-dominance gate per SOMNIA_DRAFT §FUNCTIONAL FORM.

    Condition 4 is framed as: USDT depeg + USDT/USDC basis jump.
    Per SC-2 (and the CONTEXT.md commit e600d3a literature-range-stipulation discipline),

    sc-2-usdc-literal-gate for enforcement.

    from __future__ import annotations
    from typing import Any
    import numpy as np

    # Pattern F: canonical Hawkes LL source — never call tick.score directly.
    from abrigo_x402.dgp.lr_test import _hawkes_loglik_vectorized  # noqa: F401 (loaded for downstream eval)


    REQUIRED_GATE_REPORT_KEYS: tuple[str, ...] = (
        "chainId", "contractAddress", "blockRange",
        "fetchTimestamp", "dataHash", "gitCommit", "run_id",
        "vol_of_vol_gt_zero",
        "positive_skew_fat_tails",
        "hawkes_self_excitation",
        "usdt_depeg_basis_jump",
        "any_condition_passed",
    )

    # Condition-specific thresholds (planner's discretion within CONTEXT.md envelope)
    VOL_OF_VOL_THRESHOLD: float = 0.05      # rolling-window variance dispersion floor
    KURTOSIS_EXCESS_THRESHOLD: float = 0.0  # standard fat-tail floor


    def evaluate_condition_1_vol_of_vol(rescaled_dt_per_leg: dict[str, np.ndarray]) -> dict[str, Any]:
        """Condition 1: vol-of-vol > 0 per leg.

        rescaled_dt_per_leg: {'leg_0': ndarray, 'leg_1': ndarray} from residuals.parquet.

        Test: rolling-window (window=50) variance of rescaled_dt per leg; if the std of
        those rolling variances is non-trivial (above VOL_OF_VOL_THRESHOLD) on EITHER leg,
        the condition passes.
        """
        evidence: dict[str, Any] = {"per_leg": {}}
        passed_any = False
        for leg, x in rescaled_dt_per_leg.items():
            arr = np.asarray(x, dtype=np.float64).ravel()
            window = min(50, max(10, len(arr) // 10))
            if len(arr) < window * 2:
                evidence["per_leg"][leg] = {"vol_of_vol": None, "skipped": "insufficient samples"}
                continue
            rolling_var = np.array([
                float(np.var(arr[i : i + window])) for i in range(0, len(arr) - window, window // 2)
            ])
            vov = float(np.std(rolling_var))
            evidence["per_leg"][leg] = {"vol_of_vol": vov, "window": int(window)}
            if vov > VOL_OF_VOL_THRESHOLD:
                passed_any = True
        evidence["threshold"] = VOL_OF_VOL_THRESHOLD
        return {"passed": passed_any, "evidence": evidence}


    def evaluate_condition_2_skew_fat_tails(rescaled_dt_per_leg: dict[str, np.ndarray]) -> dict[str, Any]:
        """Condition 2: positive skew AND positive excess kurtosis (fat tails) per leg.

        Test: scipy.stats.skew + scipy.stats.kurtosis (Fisher = True, so excess kurt > 0 = fat).
        Passes if EITHER leg has skew > 0 AND excess_kurt > KURTOSIS_EXCESS_THRESHOLD.
        """
        from scipy.stats import skew, kurtosis
        evidence: dict[str, Any] = {"per_leg": {}}
        passed_any = False
        for leg, x in rescaled_dt_per_leg.items():
            arr = np.asarray(x, dtype=np.float64).ravel()
            if len(arr) < 30:
                evidence["per_leg"][leg] = {"skew": None, "excess_kurt": None, "skipped": "insufficient"}
                continue
            sk = float(skew(arr))
            ek = float(kurtosis(arr, fisher=True))
            evidence["per_leg"][leg] = {"skew": sk, "excess_kurt": ek}
            if sk > 0.0 and ek > KURTOSIS_EXCESS_THRESHOLD:
                passed_any = True
        evidence["kurtosis_excess_threshold"] = KURTOSIS_EXCESS_THRESHOLD
        return {"passed": passed_any, "evidence": evidence}


    def evaluate_condition_3_hawkes_self_excitation(fit_report: dict) -> dict[str, Any]:
        """Condition 3: Hawkes self-excitation present.

        Reads Phase-3 fit_report :: hawkes_mv_params + gate_criteria. Pattern F: any
        re-evaluation MUST go through _hawkes_loglik_vectorized (imported above), NEVER
        through tick.score (wrong probability space under LS fallback).

        Passes if branching_ratio >= 0.2 AND gate_criteria['eta_floor_met'] is True.
        """
        hawkes = fit_report.get("hawkes_mv_params", {}) or {}
        branching_ratio = float(hawkes.get("branching_ratio", 0.0))
        gate_criteria = fit_report.get("gate_criteria", {}) or {}
        eta_floor_met = bool(gate_criteria.get("eta_floor_met", False))
        adjacency = np.array(hawkes.get("adjacency", [[0.0, 0.0], [0.0, 0.0]]))
        passed = (branching_ratio >= 0.2) and eta_floor_met
        return {
            "passed": passed,
            "evidence": {
                "branching_ratio": branching_ratio,
                "eta_floor_met": eta_floor_met,
                "adjacency_off_diag_max": float(np.max(np.abs(adjacency - np.diag(np.diag(adjacency))))),
                "canonical_ll_source": "_hawkes_loglik_vectorized",
            },
        }


    def evaluate_condition_4_usdt_depeg(
        calibration: dict,
        lhs_samples: np.ndarray,
        gate_decision_func: "Callable[[float, float, float], bool] | None" = None,  # type: ignore[name-defined]
    ) -> dict[str, Any]:
        """Condition 4: USDT depeg + USDT/USDC basis jump.

        Per CONTEXT.md commit e600d3a:
          evidence['source'] = 'literature_range_stipulation'  (NOT 'methodological_port',
          NOT 'Hernandez Cruz 2024' — those would be AF-03 violations per RESEARCH Pitfall 1).

        calibration: dict returned by hedge.usdt_depeg.load_calibration(). Must contain
                     'base_triple' key with {lambda_J, mu_J, sigma_J}.
        lhs_samples: (N=64, 3) array from hedge.usdt_depeg.generate_lhs_samples().
        gate_decision_func: pluggable per-cell gate decision (defaults to a permissive
                            "always pass" gate at v1.0 since the sensitivity bracket is
                            the headline; Plan 04-06 may inject a real decision rule).

        Returns: {passed, evidence: {source, base_triple, sensitivity_fragile,
                                     sensitivity_summary: {n_samples, n_flips, flip_examples}}}.
        """
        base_triple = calibration.get("base_triple", {})
        decision_func = gate_decision_func or (lambda lam, mu, sig: True)  # permissive default
        base_lam = float(base_triple.get("lambda_J", 0.05))
        base_mu = float(base_triple.get("mu_J", -0.05))
        base_sig = float(base_triple.get("sigma_J", 0.02))
        base_decision = decision_func(base_lam, base_mu, base_sig)

        flips: list[dict] = []
        for row in np.asarray(lhs_samples, dtype=np.float64):
            lam, mu, sig = float(row[0]), float(row[1]), float(row[2])
            d = decision_func(lam, mu, sig)
            if d != base_decision:
                flips.append({"lambda_J": lam, "mu_J": mu, "sigma_J": sig, "decision": d})

        n_samples = int(lhs_samples.shape[0])
        sensitivity_fragile = len(flips) > 0  # literal "any cell flip" semantics per CONTEXT.md

        return {
            "passed": bool(base_decision),
            "evidence": {
                "source": "literature_range_stipulation",
                "base_triple": {"lambda_J": base_lam, "mu_J": base_mu, "sigma_J": base_sig},
                "sensitivity_fragile": sensitivity_fragile,
                "sensitivity_summary": {
                    "n_samples": n_samples,
                    "n_flips": len(flips),
                    "flip_examples": flips[:5],  # cap the JSON payload size
                },
            },
        }
    ```
