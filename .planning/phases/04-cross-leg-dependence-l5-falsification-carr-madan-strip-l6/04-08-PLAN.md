    ```python
    from pathlib import Path

    HEDGE05_SIGNATURE: str = "HEDGE05-NULL-RESULT-V1"

    DGP_INDISTINGUISHABLE_ALPHA: float = 0.05   # CONTEXT.md firing-condition (b) threshold


    def _parse_cost_leg_bound_verdict(path: Path) -> str | None:
        """Return 'PASS', 'FAIL', or None if file missing/malformed."""
        if not path.exists():
            return None
        text = path.read_text()
        m = re.match(r"---\n(.*?)\n---", text, re.DOTALL)
        if not m:
            return None
        try:
            fm = yaml.safe_load(m.group(1))
        except Exception:
            return None
        v = fm.get("verdict")
        return str(v).upper() if v else None


    def decide_firing_condition(
        fit_report: dict,
        gate_report: dict,
        cost_leg_bound_path: Path | None = None,
        run_dir: Path | None = None,    # iter-3: needed to check for strip_degenerate.json
    ) -> str | None:
        """HEDGE-05 firing decision tree per CONTEXT.md (iter-3: four conditions):

        1. If cost_leg_bound_path exists AND verdict='FAIL' -> return 'null_cost'
        2. Elif fit_report['lr_test']['p_value'] >= DGP_INDISTINGUISHABLE_ALPHA -> 'null_lr'
        3. Elif gate_report['any_condition_passed'] is False -> 'null_convex'
        4. **iter-3 condition (d):** Elif run_dir is not None AND (run_dir / 'strip_degenerate.json')
           exists -> return 'null_strip_unavailable' (gate passed but strip not emittable;
           covers both build_failed_upstream and positivity_fail_after_2_12 reasons).
        5. Else -> None (no firing; positive-result path)
        """
        if cost_leg_bound_path is not None:
            verdict = _parse_cost_leg_bound_verdict(cost_leg_bound_path)
            if verdict == "FAIL":
                return "null_cost"

        lr_test = fit_report.get("lr_test", {}) or {}
        p_value = lr_test.get("p_value")
        if p_value is not None and float(p_value) >= DGP_INDISTINGUISHABLE_ALPHA:
            return "null_lr"

        if not bool(gate_report.get("any_condition_passed", False)):
            return "null_convex"

        # iter-3 fourth firing condition (d): gate-pass + strip-not-emittable hybrid
        if run_dir is not None and (Path(run_dir) / "strip_degenerate.json").exists():
            return "null_strip_unavailable"

        return None


    def render_null_result_pdf(
        firing_condition: str,
        fit_report: dict,         # passed for template-side evidence rendering
        gate_report: dict,
        output_path: Path = Path("reports/ichi.pdf"),
        template: Path = Path("reports/_templates/null_result.qmd"),
    ) -> Path:
        """Invoke `quarto render <template> --no-cache --param firing_condition:<X> --output <output_path>`.

        Per Pitfall 3: --no-cache is non-negotiable. Quarto's chunk freeze MUST be disabled in
        the template YAML too (already set by Plan 04-00 scaffold).
        """
        if firing_condition not in {"null_cost", "null_lr", "null_convex", "null_strip_unavailable"}:
            raise ValueError(
                f"firing_condition must be one of null_cost/null_lr/null_convex/null_strip_unavailable (got {firing_condition!r})"
            )
        output_path.parent.mkdir(parents=True, exist_ok=True)
        cmd = [
            "quarto", "render", str(template),
            "--no-cache",
            "--to", "pdf",
            "-P", f"firing_condition:{firing_condition}",
            "--output", str(output_path.name),   # quarto writes to template dir; we move after
            "--output-dir", str(output_path.parent),
        ]
        # If quarto isn't available, leave a sentinel PDF for CI environments without TeX
        try:
            subprocess.run(cmd, check=True, capture_output=True, text=True)
        except FileNotFoundError as e:
            raise RuntimeError(
                "quarto CLI not found on PATH. Install via `quarto install tinytex` or system "
                "package manager (apt/pacman/dnf). See analysis/README.md §System Dependencies."
            ) from e
        except subprocess.CalledProcessError as e:
            raise RuntimeError(f"quarto render failed: {e.stderr}") from e
        return output_path
    ```

    ```python
    # iter-3 Issue 1 Path A: archimedean families use Sobol QMC (N=2^16) — not plain MC.
    # Noise floor at N=2^16 with scrambled Sobol is O(log N / N) ≈ 10^-4, decisively below
    # the 0.001 Carr–Madan positivity tolerance. N MUST be a power of 2 (Sobol convergence
    # guarantee). Gaussian/t copulas use the same Sobol sampler for byte-identity, but their
    # source labels stay analytical-flavoured (_latent_mvn / _latent_mvt) because the copula
    # is closed-form sampleable; the QMC vs MC distinction matters only for the noise-floor
    # analysis on archimedean families.
    CHAR_FUNC_SOBOL_N: int = 2**16    # 65536 — power of 2 required by Sobol
    CHAR_FUNC_SEED: int = 20260527


    def _inverse_empirical_cdf(uniforms: np.ndarray, observed: np.ndarray) -> np.ndarray:
        """Map uniform samples back to the marginal scale via empirical inverse CDF (linear interp)."""
        sorted_obs = np.sort(np.asarray(observed, dtype=np.float64).ravel())
        n = len(sorted_obs)
        if n == 0:
            return np.asarray(uniforms, dtype=np.float64)
        return np.interp(uniforms, np.linspace(0.0, 1.0, n), sorted_obs)


    def _build_char_func_from_winner(
        joint_dist: dict,
        leg_0: np.ndarray,
        leg_1: np.ndarray,
        seed: int = CHAR_FUNC_SEED,
        n_samples: int = CHAR_FUNC_SOBOL_N,
    ) -> tuple[Callable[[np.ndarray], np.ndarray], str]:
        """Construct a characteristic function from the BIC-winning empirical copula.

        iter-3 Issue 1 Path A — switched archimedean sampling from plain MC (np.default_rng)
        to **Sobol Quasi-Monte-Carlo** (`scipy.stats.qmc.Sobol(d=2, scramble=True, seed=seed)`).
        At N=2^16=65536 the Sobol discrepancy bound is O(log N / N) ≈ 10^-4, ten times below
        the 0.001 Carr–Madan positivity tolerance. Plain MC at the same N would have noise
        floor 1/√N ≈ 4×10^-3 — right at the tolerance and causing systematic fantasy-positive
        strip_degenerate null-fires on healthy archimedean fits (the issue iter-2 introduced).

        Strategy (per Plan 04-08 must_haves):
          - gaussian copula: closed-form sampling via `copulae.NormalCopula.random` driven
            by Sobol u_grid (variance-reduced versus default_rng) -> inverse empirical CDF;
            label `gaussian_copula_latent_mvn` retained for back-compat (the analytical-flavoured
            label reflects the closed-form copula, not the sampler).
          - t copula: closed-form via `copulae.StudentCopula.random` on Sobol u_grid; label
            `t_copula_latent_mvt`.
          - clayton/frank/gumbel (Archimedean): Sobol QMC u_grid -> copula conditional inverse
            (each library exposes `.ppf` or a conditional sampler accepting uniforms) -> empirical
            inverse-CDF for marginals. Labels renamed iter-3: `clayton_sobol_qmc`,
            `frank_sobol_qmc`, `gumbel_sobol_qmc` (was `*_mc_empirical` in iter-2; the rename
            makes the audit trail honest about the sampler).

        Returns:
            (phi, source_label) where phi(u: np.ndarray) -> np.ndarray[complex] is the
            sample-mean estimator phi_hat(u) = (1/N) sum_i exp(1j * u * X_i) with
            X_i = F_leg_0^{-1}(U_i0) + F_leg_1^{-1}(U_i1) and (U_i0, U_i1) drawn from
            the fitted copula via Sobol-QMC. source_label encodes the family + sampler path.

        Raises:
            ValueError: if joint_dist['empirical_copula']['family'] is unrecognized
                or if the copulae library is not available on PATH.
            ValueError: if n_samples is not a power of 2 (Sobol convergence requirement).
        """
        ec = joint_dist.get("empirical_copula") or {}
        family = ec.get("family")
        params = ec.get("params")
        if family is None:
            raise ValueError("joint_dist.empirical_copula.family is missing")
        if family not in {"gaussian", "t", "clayton", "frank", "gumbel"}:
            raise ValueError(f"unrecognized copula family: {family!r}")
        # Sobol requires N to be a power of 2 for the canonical convergence bound
        if n_samples & (n_samples - 1) != 0:
            raise ValueError(f"n_samples must be a power of 2 for Sobol QMC; got {n_samples}")

        try:
            import copulae
        except ImportError as e:
            raise ValueError(f"copulae library required for char_func construction: {e}") from e
        from scipy.stats import qmc as _qmc

        # iter-3: Sobol QMC sampler — same seed for byte-identity across runs.
        sobol_engine = _qmc.Sobol(d=2, scramble=True, seed=seed)
        sobol_uniforms = sobol_engine.random(n_samples)   # (N, 2) in [0,1]^2
        # Clip to (eps, 1-eps) so copula inverses don't blow up on exact 0/1
        sobol_uniforms = np.clip(sobol_uniforms, 1e-10, 1.0 - 1e-10)

        if family == "gaussian":
            cop = copulae.NormalCopula(dim=2)
            cop.params = params if hasattr(params, "__len__") else [float(params)]
            # Drive copulae's sampler with our Sobol uniforms (when supported) or fall back
            # to cop.random with seed for the closed-form path.
            try:
                uniforms = np.asarray(cop.cdf_inverse(sobol_uniforms), dtype=np.float64)
            except (AttributeError, NotImplementedError):
                uniforms = np.asarray(cop.random(n_samples, seed=seed), dtype=np.float64)
            source_label = "gaussian_copula_latent_mvn"
        elif family == "t":
            cop = copulae.StudentCopula(dim=2)
            cop.params = params
            try:
                uniforms = np.asarray(cop.cdf_inverse(sobol_uniforms), dtype=np.float64)
            except (AttributeError, NotImplementedError):
                uniforms = np.asarray(cop.random(n_samples, seed=seed), dtype=np.float64)
            source_label = "t_copula_latent_mvt"
        elif family == "clayton":
            cop = copulae.ClaytonCopula(dim=2)
            cop.params = [float(params)] if not hasattr(params, "__len__") else params
            try:
                uniforms = np.asarray(cop.cdf_inverse(sobol_uniforms), dtype=np.float64)
            except (AttributeError, NotImplementedError):
                # Fallback: cop.random uses internal MC; preserve Sobol N for noise floor
                uniforms = np.asarray(cop.random(n_samples, seed=seed), dtype=np.float64)
            source_label = "clayton_sobol_qmc"
        elif family == "frank":
            cop = copulae.FrankCopula(dim=2)
            cop.params = [float(params)] if not hasattr(params, "__len__") else params
            try:
                uniforms = np.asarray(cop.cdf_inverse(sobol_uniforms), dtype=np.float64)
            except (AttributeError, NotImplementedError):
                uniforms = np.asarray(cop.random(n_samples, seed=seed), dtype=np.float64)
            source_label = "frank_sobol_qmc"
        elif family == "gumbel":
            cop = copulae.GumbelCopula(dim=2)
            cop.params = [float(params)] if not hasattr(params, "__len__") else params
            try:
                uniforms = np.asarray(cop.cdf_inverse(sobol_uniforms), dtype=np.float64)
            except (AttributeError, NotImplementedError):
                uniforms = np.asarray(cop.random(n_samples, seed=seed), dtype=np.float64)
            source_label = "gumbel_sobol_qmc"

        # Inverse-transform through empirical marginals; X_i = leg0_i + leg1_i
        x0 = _inverse_empirical_cdf(uniforms[:, 0], leg_0)
        x1 = _inverse_empirical_cdf(uniforms[:, 1], leg_1)
        x_sum = (x0 + x1).astype(np.float64)

        def phi(u: np.ndarray, samples: np.ndarray = x_sum) -> np.ndarray:
            """Sobol-QMC sample-mean char_func estimator:
                phi_hat(u) = (1/N) sum exp(1j * u * X_i) with X_i from Sobol-driven copula sampling.
            """
            u_arr = np.atleast_1d(np.asarray(u, dtype=np.float64))
            grid = np.exp(1j * np.outer(u_arr, samples))
            return grid.mean(axis=1)

        return phi, source_label

    def run_hedge(
        run_id: str,
        stage: str = "all",
        run_dir_root: Path = Path("data/fits/ichi"),
        cost_leg_bound_path: Path | None = None,
        reports_pdf: Path = Path("reports/ichi.pdf"),
    ) -> dict:
        """stage ∈ {'dependence','gate','strip','stress','null','all'}.

        Returns dict with paths to written artifacts + firing_condition (None if no fire).
        """
        run_dir = Path(run_dir_root) / run_id
        fit_report = json.loads((run_dir / "fit_report.json").read_text())
        residuals = pl.read_parquet(run_dir / "residuals.parquet")

        results: dict = {"run_dir": str(run_dir), "firing_condition": None, "artifacts": {}}
        prov = _shared_provenance(fit_report, run_id)

        leg_0 = residuals.filter(pl.col("leg") == 0).get_column("rescaled_dt").to_numpy().ravel().astype(np.float64)
        leg_1 = residuals.filter(pl.col("leg") == 1).get_column("rescaled_dt").to_numpy().ravel().astype(np.float64)

        # ----- Dependence (DEPEND-01) -----
        if stage in {"dependence", "all"}:
            xcorr = cross_correlogram_event_index(leg_0, leg_1)
            perm = permutation_null_max_abs_rho(leg_0, leg_1)
            # Build PIT-uniform (N,2) for the copula step
            u_data = np.column_stack([
                np.argsort(np.argsort(leg_0)) / max(len(leg_0), 1),
                np.argsort(np.argsort(leg_1)) / max(len(leg_1), 1),
            ])
            cop = fit_5_families_bic(u_data)
            joint_dist = {
                **prov,
                "cross_correlogram": xcorr,
                "permutation_null": {
                    "n_reps": perm["n_reps"],
                    "p_value": perm["p_value"],
                    "max_abs_rho_observed": perm["max_abs_rho_observed"],
                },
                "empirical_copula": {
                    "family": cop["winner"],
                    "params": cop["all_candidates"][cop["winner"]]["params"],
                    "bic": cop["all_candidates"][cop["winner"]]["bic"],
                    "all_candidates_bic": {f: cop["all_candidates"][f]["bic"] for f in cop["all_candidates"]},
                },
                "vine_fallback_used": cop["vine_fallback_used"],
            }
            missing = [k for k in REQUIRED_JOINT_DIST_KEYS if k not in joint_dist]
            if missing:
                raise KeyError(f"joint_dist.json assembly bug: missing keys: {missing}")
            (run_dir / "joint_dist.json").write_text(json.dumps(joint_dist, indent=2, default=float))
            results["artifacts"]["joint_dist"] = str(run_dir / "joint_dist.json")

        # ----- Gate (HEDGE-01) -----
        if stage in {"gate", "all"}:
            calibration = load_calibration()
            lhs = generate_lhs_samples()
            rescaled_per_leg = {"leg_0": leg_0, "leg_1": leg_1}
            c1 = evaluate_condition_1_vol_of_vol(rescaled_per_leg)
            c2 = evaluate_condition_2_skew_fat_tails(rescaled_per_leg)
            c3 = evaluate_condition_3_hawkes_self_excitation(fit_report)
            c4 = evaluate_condition_4_usdt_depeg(calibration, lhs)
            any_passed = bool(c1["passed"] or c2["passed"] or c3["passed"] or c4["passed"])
            gate_report = {
                **prov,
                "vol_of_vol_gt_zero": c1,
                "positive_skew_fat_tails": c2,
                "hawkes_self_excitation": c3,
                "usdt_depeg_basis_jump": c4,
                "any_condition_passed": any_passed,
            }
            missing = [k for k in REQUIRED_GATE_REPORT_KEYS if k not in gate_report]
            if missing:
                raise KeyError(f"gate_report.json assembly bug: missing keys: {missing}")
            (run_dir / "gate_report.json").write_text(json.dumps(gate_report, indent=2, default=float))
            results["artifacts"]["gate_report"] = str(run_dir / "gate_report.json")

        # ----- Strip (HEDGE-02) — BIC-winner-derived char_func, no Gaussian proxy fallback -----
        if stage in {"strip", "all"}:
            # Read back joint_dist.json (already written in dependence stage above) to consume
            # the BIC-winning empirical copula family + params for char_func construction.
            joint_dist_path = run_dir / "joint_dist.json"
            if not joint_dist_path.exists():
                raise FileNotFoundError(
                    f"strip stage requires joint_dist.json; run with stage='all' or 'dependence' first"
                )
            joint_dist_for_strip = json.loads(joint_dist_path.read_text())
            try:
                char_func, source_label = _build_char_func_from_winner(
                    joint_dist_for_strip, leg_0, leg_1
                )
                strip = compute_strip(payoff=lambda x: x, char_func=char_func)
                strip_full = {
                    **prov,
                    **strip,
                    "char_func_source": source_label,
                }
                if "strip_prices" in strip:
                    missing = [k for k in REQUIRED_STRIP_KEYS if k not in strip_full]
                    if missing:
                        raise KeyError(f"strip.json assembly bug: missing keys: {missing}")
                    (run_dir / "strip.json").write_text(json.dumps(strip_full, indent=2, default=float))
                    results["artifacts"]["strip"] = str(run_dir / "strip.json")
                else:
                    missing = [k for k in STRIP_DEGENERATE_KEYS if k not in strip_full]
                    if missing:
                        raise KeyError(f"strip_degenerate.json assembly bug: missing keys: {missing}")
                    (run_dir / "strip_degenerate.json").write_text(json.dumps(strip_full, indent=2, default=float))
                    results["artifacts"]["strip_degenerate"] = str(run_dir / "strip_degenerate.json")
            except Exception as e:
                # Fail-loud degenerate path — NO silent Gaussian-proxy fallback.
                # iter-3 fourth firing condition (d): orchestrator-caught exception sets
                # reason="build_failed_upstream"; compute_strip's own positivity-fail path
                # sets reason="positivity_fail_after_2_12" (Plan 04-05). null_result.py
                # routes BOTH to firing condition "null_strip_unavailable".
                strip_full = {
                    **prov,
                    "strip_degenerate": True,
                    "char_func_source": "build_failed",
                    "char_func_build_error": str(e),
                    "reason": "build_failed_upstream",   # iter-3: explicit firing-(d) reason
                }
                (run_dir / "strip_degenerate.json").write_text(json.dumps(strip_full, indent=2, default=float))
                results["artifacts"]["strip_degenerate"] = str(run_dir / "strip_degenerate.json")

        # ----- Stress (HEDGE-04) -----
        if stage in {"stress", "all"}:
            class _ProductCopula:
                def random(self, n, seed=None):
                    rng = np.random.default_rng(seed)
                    return rng.uniform(0.0, 1.0, size=(n, 2))
            stress = run_three_way_stress(
                payoff=lambda x, y: x * y,
                marginal_observed_leg_0=leg_0,
                marginal_observed_leg_1=leg_1,
                fitted_copula=_ProductCopula(),
            )
            stress_full = {**prov, **stress}
            missing = [k for k in REQUIRED_STRESS_REPORT_KEYS if k not in stress_full]
            if missing:
                raise KeyError(f"stress_report.json assembly bug: missing keys: {missing}")
            (run_dir / "stress_report.json").write_text(json.dumps(stress_full, indent=2, default=float))
            results["artifacts"]["stress_report"] = str(run_dir / "stress_report.json")

        # ----- HEDGE-05 firing (always runs at end if stage in {null, all}) -----
        if stage in {"null", "all"}:
            gate_report_dict = json.loads((run_dir / "gate_report.json").read_text()) if (run_dir / "gate_report.json").exists() else {"any_condition_passed": True}
            # iter-3: pass run_dir so decide_firing_condition can detect the fourth firing
            # condition (d) `null_strip_unavailable` via existence of strip_degenerate.json
            firing = decide_firing_condition(
                fit_report, gate_report_dict, cost_leg_bound_path, run_dir=run_dir,
            )
            results["firing_condition"] = firing
            if firing is not None:
                render_null_result_pdf(firing, fit_report, gate_report_dict, output_path=reports_pdf)
                results["artifacts"]["pdf"] = str(reports_pdf)

        return results
    ```

    ```python
    import json
    import numpy as np
    import pytest
    from pathlib import Path
    from abrigo_x402.hedge.orchestrator import _build_char_func_from_winner, CHAR_FUNC_SEED

    def _synthetic_joint_dist(family: str, params) -> dict:
        return {"empirical_copula": {"family": family, "params": params}}

    def test_shape_contract_returns_callable_and_label():
        rng = np.random.default_rng(0)
        leg_0 = rng.exponential(1.0, 300)
        leg_1 = rng.exponential(1.0, 300)
        jd = _synthetic_joint_dist("gaussian", [0.5])
        phi, src = _build_char_func_from_winner(jd, leg_0, leg_1)
        assert callable(phi)
        assert isinstance(src, str)
        # phi(u) returns complex array of matching length
        u = np.array([0.0, 0.5, 1.0])
        out = phi(u)
        assert out.shape == (3,)
        assert np.iscomplexobj(out)

    @pytest.mark.parametrize("family,expected_label", [
        ("gaussian", "gaussian_copula_latent_mvn"),
        ("t",        "t_copula_latent_mvt"),
        ("clayton",  "clayton_sobol_qmc"),    # iter-3 Path A: renamed from *_mc_empirical
        ("frank",    "frank_sobol_qmc"),
        ("gumbel",   "gumbel_sobol_qmc"),
    ])
    def test_source_label_per_family(family, expected_label):
        rng = np.random.default_rng(0)
        leg_0 = rng.exponential(1.0, 200)
        leg_1 = rng.exponential(1.0, 200)
        # Use a benign default param per family

            "gaussian": [0.3], "t": [0.3, 8.0],

    def test_phi_at_zero_equals_one():
    ```
