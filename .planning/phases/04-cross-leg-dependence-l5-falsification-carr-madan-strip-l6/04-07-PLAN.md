    Implement the three scenarios as sample-based price estimates (since `compute_strip` operates on characteristic functions, the simpler v1.0 path is Monte-Carlo: draw N samples per scenario from the appropriate joint, push through `payoff`, average).

    ```python
    """Three-way joint-distribution stress: independence + fitted_joint + comonotone (Frechet upper bound).

    Per CONTEXT.md HEDGE-04: flag-only at >30%, no hard-fail.
    Per RESEARCH Pitfall 6: comonotone uses 'empirical_body_parametric_tail' method label.
    Per RESEARCH §"Don't Hand-Roll": comonotone via shared U sampling.
    """
    from __future__ import annotations
    from typing import Callable
    import numpy as np

    REQUIRED_STRESS_REPORT_KEYS: tuple[str, ...] = (
        "chainId", "contractAddress", "blockRange",
        "fetchTimestamp", "dataHash", "gitCommit", "run_id",
        "independence_price", "fitted_joint_price", "comonotone_price",
        "divergence_pct", "divergence_flag", "comonotone_method",
    )

    DIVERGENCE_FLAG_THRESHOLD_PCT: float = 30.0   # CONTEXT.md lock — flag-only


    def _price_under_joint_samples(
        payoff: Callable[[np.ndarray, np.ndarray], np.ndarray],
        samples: np.ndarray,   # (N, 2) joint samples on the original scale
    ) -> float:
        """Monte-Carlo price = E[payoff(x_0, x_1)] under the joint sample."""
        vals = payoff(samples[:, 0], samples[:, 1])
        return float(np.mean(vals))


    def _inverse_empirical_cdf(uniforms: np.ndarray, observed: np.ndarray) -> np.ndarray:
        """Map uniform samples back to the marginal scale via empirical inverse CDF."""
        sorted_obs = np.sort(observed)
        n = len(sorted_obs)
        if n == 0:
            return uniforms
        # Linear interpolation on the sorted observed (empirical body), edges fall back to extremes.
        return np.interp(uniforms, np.linspace(0.0, 1.0, n), sorted_obs)


    def run_three_way_stress(
        payoff: Callable[[np.ndarray, np.ndarray], np.ndarray],
        marginal_observed_leg_0: np.ndarray,
        marginal_observed_leg_1: np.ndarray,
        fitted_copula,                    # has .random(n, seed=...) -> (n, 2) uniforms
        n_samples: int = 10_000,
        seed: int = 20260527,
    ) -> dict:
        """Compute strip-equivalent E[payoff] under three joint scenarios + divergence flag.

        Scenarios:
          1. independence: U_1, U_2 ~ U(0,1) independently
          2. fitted_joint: fitted_copula.random(N, seed=seed)
          3. comonotone:  U_1 ~ U(0,1), U_2 = U_1 (Frechet upper bound)
        Per-scenario: inverse-transform via empirical CDF of observed marginals
        ('empirical_body_parametric_tail' label per RESEARCH Pitfall 6 — v1.0 uses empirical body;
        Plan 04-08 may extend to parametric tail if BIC winner provides marginal model).
        """
        rng = np.random.default_rng(seed)

        # 1. Independence
        u_ind = rng.uniform(0.0, 1.0, size=(n_samples, 2))
        x_ind_0 = _inverse_empirical_cdf(u_ind[:, 0], marginal_observed_leg_0)
        x_ind_1 = _inverse_empirical_cdf(u_ind[:, 1], marginal_observed_leg_1)
        independence_price = _price_under_joint_samples(payoff, np.column_stack([x_ind_0, x_ind_1]))

        # 2. Fitted joint
        u_fit = np.asarray(fitted_copula.random(n_samples, seed=seed), dtype=np.float64)
        x_fit_0 = _inverse_empirical_cdf(u_fit[:, 0], marginal_observed_leg_0)
        x_fit_1 = _inverse_empirical_cdf(u_fit[:, 1], marginal_observed_leg_1)
        fitted_joint_price = _price_under_joint_samples(payoff, np.column_stack([x_fit_0, x_fit_1]))

        # 3. Comonotone (Frechet upper bound: U_2 = U_1)
        u_com = rng.uniform(0.0, 1.0, size=n_samples)
        x_com_0 = _inverse_empirical_cdf(u_com, marginal_observed_leg_0)
        x_com_1 = _inverse_empirical_cdf(u_com, marginal_observed_leg_1)
        comonotone_price = _price_under_joint_samples(payoff, np.column_stack([x_com_0, x_com_1]))

        # Divergence flag
        prices = np.array([independence_price, fitted_joint_price, comonotone_price])
        mean_price = float(np.mean(prices))
        if mean_price == 0.0:
            divergence_pct = 0.0
        else:
            divergence_pct = float((np.max(prices) - np.min(prices)) / abs(mean_price) * 100.0)
        divergence_flag = bool(divergence_pct > DIVERGENCE_FLAG_THRESHOLD_PCT)

        return {
            "independence_price": float(independence_price),
            "fitted_joint_price": float(fitted_joint_price),
            "comonotone_price": float(comonotone_price),
            "divergence_pct": divergence_pct,
            "divergence_flag": divergence_flag,
            "comonotone_method": "empirical_body_parametric_tail",
        }
    ```
