    ```python
    """USDT depeg jump-leg calibration: literature-range stipulation + N=64 LHS sensitivity.

    Per CONTEXT.md commit e600d3a: evidence_source='literature_range_stipulation'.
    Hernandez Cruz 2024 and Wu & Liu 2026 do NOT publish jump-diffusion parameters
    and MUST NOT be cited as parameter sources (RESEARCH Pitfall 1).
    """
    from __future__ import annotations
    import re
    import numpy as np
    import yaml
    from pathlib import Path
    from scipy.stats import qmc

    DEFAULT_LAMBDA_J: float = 0.05
    DEFAULT_MU_J: float = -0.05
    DEFAULT_SIGMA_J: float = 0.02
    LHS_N_SAMPLES: int = 64
    LHS_BOUND_RATIO: float = 0.5
    LHS_SEED: int = 20260527


    def load_calibration(calibration_path: str | Path = "notes/usdt_depeg_calibration.md") -> dict:
        """Parse YAML frontmatter; return {evidence_source, base_triple: {lambda_J, mu_J, sigma_J}}."""
        text = Path(calibration_path).read_text()
        m = re.match(r"---\n(.*?)\n---", text, re.DOTALL)
        if not m:
            raise ValueError(f"{calibration_path}: missing YAML frontmatter delimited by ---")
        fm = yaml.safe_load(m.group(1))
        if fm.get("evidence_source") != "literature_range_stipulation":
            raise ValueError(
                f"{calibration_path}: evidence_source must be 'literature_range_stipulation' "
                f"per CONTEXT.md commit e600d3a (got {fm.get('evidence_source')!r})"
            )
        bt = fm.get("base_triple", {})
        return {
            "evidence_source": fm["evidence_source"],
            "base_triple": {
                "lambda_J": float(bt.get("lambda_J", DEFAULT_LAMBDA_J)),
                "mu_J": float(bt.get("mu_J", DEFAULT_MU_J)),
                "sigma_J": float(bt.get("sigma_J", DEFAULT_SIGMA_J)),
            },
        }


    def generate_lhs_samples(
        base_lambda_J: float = DEFAULT_LAMBDA_J,
        base_mu_J: float = DEFAULT_MU_J,
        base_sigma_J: float = DEFAULT_SIGMA_J,
        n_samples: int = LHS_N_SAMPLES,
        bound_ratio: float = LHS_BOUND_RATIO,
        seed: int = LHS_SEED,
    ) -> np.ndarray:
        """Return (n_samples, 3) float64 array of (lambda_J, mu_J, sigma_J) LHS samples.

        Bounds: per-dimension [base*(1-bound_ratio), base*(1+bound_ratio)], with bounds order
        normalized via min/max to handle negative bases (e.g., mu_J).
        """
        sampler = qmc.LatinHypercube(d=3, seed=seed)
        unit = sampler.random(n=n_samples)
        bases = np.array([base_lambda_J, base_mu_J, base_sigma_J], dtype=np.float64)
        lo = np.minimum(bases * (1 - bound_ratio), bases * (1 + bound_ratio))
        hi = np.maximum(bases * (1 - bound_ratio), bases * (1 + bound_ratio))
        scaled = qmc.scale(unit, l_bounds=lo, u_bounds=hi)
        return scaled.astype(np.float64)
    ```
