    ```typescript
    import type { ProtocolSpec } from './protocol-spec';

    export interface BudgetEstimate {
      protocol: string;
      iteration: number;
      vault_count: number;
      blocks_per_vault: number;
      queries_per_vault: number;
      total_queries: number;
      exceeds_earmark: boolean;
      earmark: number;
      recommended_reallocation: string | null;
    }

    /**
     * Per-vault query model (RESEARCH §M):
     *   - subgraph LP-fee queries: ~1 per 10k blocks (Phase 1 default DORMANT; counted hypothetically here)
     *   - Blockscout topic-filtered pulls: ~1 per 1000 swaps (~5 for cKES/USDT @ 4400 swaps/30d)
     */
    export function estimateBudget(spec: ProtocolSpec, fornoHead: number, earmark = 30_000): BudgetEstimate {
      const activeVaults = Object.entries(spec.vaults).filter(([_, v]) => v.active);
      const vault_count = activeVaults.length;

      const fromBlock = spec.cold_backfill_from_block ?? 65_000_000;
      const blocks_per_vault = Math.max(0, fornoHead - fromBlock);

      const subgraph_queries_per_vault = Math.ceil(blocks_per_vault / 10_000);
      const observed_swaps_30d = spec.anchor_pool.swaps_per_30d_observed ?? 4_400;
      const blockscout_queries_per_vault = Math.ceil(observed_swaps_30d / 1_000);
      const queries_per_vault = subgraph_queries_per_vault + blockscout_queries_per_vault;

      const total_queries = vault_count * queries_per_vault;
      const graph_total = vault_count * subgraph_queries_per_vault;
      const exceeds_earmark = graph_total > earmark;

      return {
        protocol: spec.name,
        iteration: spec.iteration,
        vault_count,
        blocks_per_vault,
        queries_per_vault,
        total_queries,
        exceeds_earmark,
        earmark,
        recommended_reallocation: exceeds_earmark
          ? `Projected graph queries (${graph_total}) exceeds ${earmark} earmark. Options: (a) re-scope vault set, (b) reallocate from 45k reserve, (c) pass --force to bypass at the budget gate.`
          : null,
      };
    }
    ```

    ```typescript
    #!/usr/bin/env tsx
    import { readFileSync } from 'node:fs';
    import { fileURLToPath } from 'node:url';
    import { dirname, join } from 'node:path';
    import { loadProtocol } from './protocol-spec';
    import { celoClient } from './viem-clients';
    import { estimateBudget } from './budget';
    import { checkBudget, GraphBudgetExceededError } from './cost-ledger';
    import { cacheKeyHash } from './cache/key';
    import { findByCacheKey } from './cache/manifest';

    const __dirname = dirname(fileURLToPath(import.meta.url));
    const pkg = JSON.parse(readFileSync(join(__dirname, '..', 'package.json'), 'utf-8'));

    function parseArgs(argv: string[]) {
      const flags: Record<string, string | boolean> = {};
      const positional: string[] = [];
      for (let i = 0; i < argv.length; i++) {
        const a = argv[i];
        if (a.startsWith('--')) {
          const name = a.slice(2);
          const next = argv[i + 1];
          if (next && !next.startsWith('--')) {
            flags[name] = next; i++;
          } else {
            flags[name] = true;
          }
        } else {
          positional.push(a);
        }
      }
      return { flags, positional };
    }

    async function main() {
      const argv = process.argv.slice(2);
      if (argv.includes('--version') || argv.includes('-v')) {
        console.log(pkg.version); return 0;
      }
      const { flags, positional } = parseArgs(argv);
      const cmd = positional[0];
      if (cmd !== 'ichi' && cmd !== 'steer') {
        console.error('Usage: pnpm -C fetch fetch <ichi|steer> [--dry-run] [--estimate-budget] [--force] [--pool 0x...] [--from N] [--to N]');
        return 1;
      }

      const specPath = `../protocols/${cmd}.toml`;
      const spec = await loadProtocol(specPath);

      // Determine head + record source (checker C3 — dry-run NEVER calls Forno).
      // Priority: env override > committed snapshot > constants.ts fallback (dry-run only) > live Forno (non-dry-run).
      let head: number;
      let head_source: 'env_override' | 'snapshot' | 'fallback' | 'live';
      if (process.env.FETCH_HEAD_OVERRIDE) {
        head = Number(process.env.FETCH_HEAD_OVERRIDE);
        head_source = 'env_override';
      } else if (flags['dry-run']) {
        // Dry-run must be deterministic + network-free in CI.
        try {
          const { loadFornoHeadSnapshot } = await import('./constants');
          const snap = loadFornoHeadSnapshot();
          head = snap.head;
          head_source = 'snapshot';
        } catch {
          const { DRY_RUN_FALLBACK_HEAD } = await import('./constants');
          head = DRY_RUN_FALLBACK_HEAD;
          head_source = 'fallback';
        }
      } else {
        head = Number(await celoClient.getBlockNumber());
        head_source = 'live';
      }

      const est = estimateBudget(spec, head);

      // Cache short-circuit check (if --pool/--from/--to provided)
      let cache_hit = false;
      if (flags.pool && flags.from && flags.to) {
        const key = cacheKeyHash({
          chainId: spec.chain_id,
          contractAddress: String(flags.pool),
          blockRange: [Number(flags.from), Number(flags.to)],
        });
        const existing = await findByCacheKey(key);
        cache_hit = existing !== null;
      }

      // Budget gate (only if NOT pure dry-run)
      if (!flags['dry-run']) {
        try {
          await checkBudget({
            projected_graph_queries: est.vault_count * Math.ceil(est.blocks_per_vault / 10_000),
            force: Boolean(flags.force),
          });
        } catch (e) {
          if (e instanceof GraphBudgetExceededError) {
            console.error(e.message);
            return 2;
          }
          throw e;
        }
      }

      const output = {
        ...est,
        cache_hit,
        dry_run: Boolean(flags['dry-run']),
        head_block: head,
        head_source,
      };
      if (flags['estimate-budget'] || flags['dry-run']) {
        console.log(JSON.stringify(output, null, 2));
      }
      return 0;
    }

    main().then((code) => process.exit(code)).catch((err) => {
      console.error('FATAL:', err.message);
      process.exit(99);
    });
    ```

    ```typescript
    import { execSync } from 'node:child_process';

    function runCli(args: string, env: Record<string, string> = {}): { stdout: string; stderr: string; code: number } {
      try {
        const stdout = execSync(`pnpm -s exec tsx src/cli.ts ${args}`, {

          encoding: 'utf-8',
          env: { ...process.env, FETCH_HEAD_OVERRIDE: '67854122', ...env },

        return { stdout, stderr: '', code: 0 };
      } catch (e: any) {
        return { stdout: e.stdout?.toString() ?? '', stderr: e.stderr?.toString() ?? '', code: e.status ?? 1 };
      }
    }

    describe('CLI integration', () => {
      test('--version exits 0 with semver', () => {
    ```
