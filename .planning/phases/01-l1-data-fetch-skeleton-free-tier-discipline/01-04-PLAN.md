    ```typescript
    import { createHash } from 'node:crypto';

    export interface CacheKeyInput {
      chainId: number;
      contractAddress: string;
      blockRange: [number, number];
    }

    export function canonicalize({ chainId, contractAddress, blockRange }: CacheKeyInput): string {
      // CRITICAL (checker C2): use an explicit string template literal — NOT JSON.stringify
      // with an array second-arg, which is a PROPERTY FILTER and does NOT enforce key order.
      // The previous implementation was byte-stable only by accident of V8 insertion-order
      // preservation; this version is mechanically byte-stable.
      const addr = contractAddress.toLowerCase();
      const lo = blockRange[0];
      const hi = blockRange[1];
      // Alphabetical key order: blockRange, chainId, contractAddress.
      // No whitespace. Numbers are stringified by JS's default (no trailing .0 for ints).
      return `{"blockRange":[${lo},${hi}],"chainId":${chainId},"contractAddress":"${addr}"}`;
    }

    export function cacheKeyHash(input: CacheKeyInput): string {
      return createHash('sha256').update(canonicalize(input)).digest('hex');
    }
    ```

    ```typescript
    import { readFile, writeFile, mkdir } from 'node:fs/promises';
    import { existsSync } from 'node:fs';
    import { dirname } from 'node:path';
    import { z } from 'zod';

    export const ManifestEntrySchema = z.object({
      cache_key_hash: z.string().regex(/^[0-9a-f]{64}$/),
      chainId: z.number().int(),
      contractAddress: z.string(),
      blockRange: z.tuple([z.number(), z.number()]),
      fetchTimestamp: z.string(),
      dataHash: z.string().regex(/^[0-9a-f]{64}$/),
      gitCommit: z.string(),
      endpoint: z.enum(['graph-mainnet', 'graph-sepolia', 'blockscout', 'forno', 'x402-mock-sepolia']),
      query_count: z.number().int().nonnegative(),
      usdc_cost: z.string(),
      paid_real: z.boolean(),
      protocol: z.string(),
      fetcher_version: z.string(),
    });
    export type ManifestEntry = z.infer<typeof ManifestEntrySchema>;

    const DEFAULT_MANIFEST = 'data/raw/manifest.json';

    export async function readManifest(path: string = DEFAULT_MANIFEST): Promise<ManifestEntry[]> {
      if (!existsSync(path)) return [];
      const text = await readFile(path, 'utf-8');
      if (text.trim().length === 0) return [];
      const arr = JSON.parse(text);
      return z.array(ManifestEntrySchema).parse(arr);
    }

    export async function appendManifestIfNew(
      entry: ManifestEntry,
      path: string = DEFAULT_MANIFEST
    ): Promise<{ appended: boolean; existing?: ManifestEntry }> {
      ManifestEntrySchema.parse(entry);
      const existing = await readManifest(path);
      const dup = existing.find((e) => e.cache_key_hash === entry.cache_key_hash);
      if (dup) return { appended: false, existing: dup };
      const next = [...existing, entry];
      if (!existsSync(dirname(path))) await mkdir(dirname(path), { recursive: true });
      // Write to tmp then rename (atomic)
      const tmp = `${path}.tmp`;
      await writeFile(tmp, JSON.stringify(next, null, 2));
      const { rename } = await import('node:fs/promises');
      await rename(tmp, path);
      return { appended: true };
    }

    export async function findByCacheKey(
      cache_key_hash: string,
      path: string = DEFAULT_MANIFEST
    ): Promise<ManifestEntry | null> {
      const all = await readManifest(path);
      return all.find((e) => e.cache_key_hash === cache_key_hash) ?? null;
    }
    ```

    **fetch/src/cache/parquet-writer.ts** — Phase 1 ships deterministic JSONL (Phase 2 batch-converts to Parquet):
    ```typescript
    import { writeFile, mkdir } from 'node:fs/promises';
    import { existsSync } from 'node:fs';
    import { createHash } from 'node:crypto';
    import { join } from 'node:path';

    /**
     * Phase 1: writes deterministic JSONL. Phase 2 polars-ingest converts to Parquet.
     * Determinism: rows are written verbatim in caller's order; no Date.now() injected; no embedded fetch timestamp.
     * dataHash = sha256(file_content) so two identical invocations produce identical files AND identical hashes.
     */
    export async function writeCachePayload(
      rows: Record<string, unknown>[],
      opts: { cache_key_hash: string; protocol: string; root?: string }
    ): Promise<{ path: string; dataHash: string }> {
      const { cache_key_hash, protocol, root = 'data/raw' } = opts;
      const prefix = cache_key_hash.slice(0, 2);
      const dir = join(root, protocol, prefix);
      if (!existsSync(dir)) await mkdir(dir, { recursive: true });
      const path = join(dir, `${cache_key_hash}.jsonl`);
      // Stable serialization: JSON.stringify per row in input order, no whitespace, newline-delimited
      const content = rows.map((r) => JSON.stringify(r)).join('\n') + (rows.length > 0 ? '\n' : '');
      await writeFile(path, content);
      const dataHash = createHash('sha256').update(content).digest('hex');
      return { path, dataHash };
    }
    ```
