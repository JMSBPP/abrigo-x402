    **fetch/src/blockscout/v1-getlogs.ts**:
    ```typescript
    import { z } from 'zod';

    const LogSchema = z.object({
      address: z.string(),
      blockNumber: z.string(),
      timeStamp: z.string(),
      logIndex: z.string(),
      transactionHash: z.string(),
      topics: z.array(z.string().nullable()),
      data: z.string(),
    });
    const ResponseSchema = z.object({
      status: z.string(),
      message: z.string(),
      result: z.array(LogSchema),
    });
    export type BlockscoutLog = z.infer<typeof LogSchema>;

    export async function getLogsV1(opts: {
      baseUrl: string;
      address: string;
      fromBlock: number;
      toBlock: number;
      topic0: string;
      apiKey?: string;
      fetcher?: typeof fetch;
    }): Promise<BlockscoutLog[]> {
      const { baseUrl, address, fromBlock, toBlock, topic0, apiKey, fetcher = fetch } = opts;
      const all: BlockscoutLog[] = [];
      let cursor = fromBlock;
      while (cursor <= toBlock) {
        const url = new URL(`${baseUrl}/api`);
        url.searchParams.set('module', 'logs');
        url.searchParams.set('action', 'getLogs');
        url.searchParams.set('address', address);
        url.searchParams.set('fromBlock', String(cursor));
        url.searchParams.set('toBlock', String(toBlock));
        url.searchParams.set('topic0', topic0);
        if (apiKey) url.searchParams.set('apikey', apiKey);

        const res = await fetcher(url.toString());
        if (!res.ok) throw new Error(`Blockscout v1 getLogs HTTP ${res.status}`);
        const parsed = ResponseSchema.parse(await res.json());
        if (parsed.status !== '1' && parsed.result.length === 0) {
          // "No records found" is a valid empty response in v1 etherscan-compat
          if (parsed.message?.includes('No records')) break;
          throw new Error(`Blockscout v1 status=${parsed.status}: ${parsed.message}`);
        }
        all.push(...parsed.result);

        if (parsed.result.length < 1000) break;        // last page
        const lastBlockHex = parsed.result[parsed.result.length - 1].blockNumber;
        cursor = parseInt(lastBlockHex, 16) + 1;
      }
      return all;
    }
    ```

    ```typescript
    import { GraphQLClient } from 'graphql-request';

    export class MissingGraphApiKeyError extends Error {
      constructor(msg?: string) {
        super(msg ?? 'GRAPH_API_KEY env var is unset — subgraph leg disabled. Phase 1 default is Blockscout-only per RESEARCH §A.');
        this.name = 'MissingGraphApiKeyError';
      }
    }

    export function getSubgraphClient(deployment_id: string): GraphQLClient {
      // Read env at CALL TIME (not module-load time) per checker I8 — enables vi.stubEnv tests.
      const apiKey = process.env.GRAPH_API_KEY;
      if (!apiKey || apiKey.length === 0) {
        throw new MissingGraphApiKeyError();
      }
      const url = `https://gateway.thegraph.com/api/${apiKey}/subgraphs/id/${deployment_id}`;
      return new GraphQLClient(url);
    }
    ```
