```python
"""DGP-03: boundary-correct parametric bootstrap LR test for NHPP vs bivariate Hawkes.

Reference: Cavaliere et al. 2022 arxiv:2104.03122 (fixed intensity bootstrap);
           Filimonov & Sornette 2014 arxiv:1403.5227 (boundary correction);
           Self & Liang 1987 (asymptotic mixture theory).

LOCKED INVARIANTS (PRE_REGISTRATION.md §Test Statistics):
- n_reps = 1000 in production (PRE_REGISTRATION lock; --bootstrap-reps is dev-only override)
- alpha = 0.01 single-tier headline
- Null distribution = 50:50 χ²(0):χ²(1) mixture (empirically realized via bootstrap)
- Deterministic seed = sha256(panel_dataHash + b"phase-3-bootstrap")[:4] as uint32

FORBIDDEN (SC-3 grep gate):
- `from scipy.stats import chi2; ...chi2(1).sf(...)` — wrong null distribution
- `from statsmodels.stats.diagnostic import likelihood_ratio_test` — boundary-blind

The bootstrap MUST simulate from the fitted NHPP (Pitfall 2), NOT from the fitted Hawkes.
"""
from __future__ import annotations
import hashlib
import os
from pathlib import Path

import numpy as np

# Headless matplotlib — must be set BEFORE pyplot import
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402

from tick.hawkes import SimuHawkesExpKernels

from abrigo_x402.dgp.nhpp_inar import fit_nhpp_inar
from abrigo_x402.dgp.hawkes_fit import fit_hawkes_expkern

PRODUCTION_N_REPS: int = 1000
DEFAULT_ALPHA: float = 0.01


def _derive_seed(panel_data_hash: str) -> int:
    """sha256(panel_dataHash + 'phase-3-bootstrap')[:4] as uint32 — CONTEXT.md decision."""
    digest = hashlib.sha256((panel_data_hash + "phase-3-bootstrap").encode()).digest()
    return int.from_bytes(digest[:4], "big")


def _simulate_nhpp_under_null(
    nhpp_baseline_per_sec: np.ndarray,  # shape (2,) — events/sec rate per leg from fitted INAR(p)
    end_time: float,
    seed: int,
) -> tuple[np.ndarray, np.ndarray]:
    """Simulate bivariate inhomogeneous Poisson via tick with adjacency=zeros (Research Open Q 2).

    Setting adjacency to a zero 2x2 matrix reduces SimuHawkesExpKernels to a bivariate
    inhomogeneous Poisson process with the supplied baseline rate vector — i.e., pure NHPP.

    NEVER set force_simulation=True (Pitfall 9).
    """
    sim = SimuHawkesExpKernels(
        adjacency=np.zeros((2, 2), dtype=np.float64),
        decays=0.1,  # ignored when adjacency=0
        baseline=nhpp_baseline_per_sec.astype(np.float64),
        end_time=float(end_time),
        seed=int(seed),
        verbose=False,
    )
    sim.simulate()
    return sim.timestamps[0].astype(np.float64), sim.timestamps[1].astype(np.float64)


def _score_nhpp_loglik(nhpp_fit: dict, leg_0: np.ndarray, leg_1: np.ndarray) -> float:
    """Return in-sample log-likelihood already stored on the fit dict."""
    return float(nhpp_fit["loglik_in_sample"])


def _nhpp_baseline_per_sec(nhpp_fit: dict) -> np.ndarray:
    """Kirchner scaling: per-bin intercept / bin_width = events/sec rate per leg."""
    intercept = np.asarray(nhpp_fit["intercept"], dtype=np.float64)
    bw = float(nhpp_fit["bin_width_seconds"])
    return intercept / bw


def parametric_bootstrap_lr(
    leg_0_times: np.ndarray,
    leg_1_times: np.ndarray,
    panel_data_hash: str,
    window_start: float,
    window_end: float,
    n_reps: int = PRODUCTION_N_REPS,
    alpha: float = DEFAULT_ALPHA,
    diagnostic_plot_path: str | os.PathLike | None = None,
) -> dict:
    """Boundary-correct parametric bootstrap LR test for NHPP-vs-Hawkes.

    Returns dict with keys: observed_stat, bootstrap_null_dist_50_50_chi2_0_chi2_1, p_value,
    rejects_at_alpha, n_reps, n_failed, seed, alpha.
    """
    seed = _derive_seed(panel_data_hash)
    rng = np.random.default_rng(seed)

    # 1. Fit BOTH models on observed data
    nhpp_obs = fit_nhpp_inar(leg_0_times, leg_1_times, window_start, window_end)
    hawkes_obs = fit_hawkes_expkern(leg_0_times, leg_1_times)
    LL_nhpp_obs = _score_nhpp_loglik(nhpp_obs, leg_0_times, leg_1_times)
    LL_hawkes_obs = float(hawkes_obs["loglik_in_sample"])
    LR_observed = 2.0 * (LL_hawkes_obs - LL_nhpp_obs)

    # 2. Parametric bootstrap UNDER THE NULL (NHPP, NOT Hawkes — Pitfall 2)
    nhpp_baseline = _nhpp_baseline_per_sec(nhpp_obs)
    bootstrap_LR = np.full(n_reps, np.nan, dtype=np.float64)
    n_failed = 0
    for k in range(n_reps):
        rep_seed = int(rng.integers(0, 2**31))
        try:
            sim_0, sim_1 = _simulate_nhpp_under_null(nhpp_baseline, window_end - window_start, rep_seed)
            if sim_0.size < 10 or sim_1.size < 10:
                n_failed += 1
                continue
            # Shift simulated times back to absolute window
            sim_0 = sim_0 + window_start
            sim_1 = sim_1 + window_start
            nhpp_b = fit_nhpp_inar(sim_0, sim_1, window_start, window_end)
            hawkes_b = fit_hawkes_expkern(sim_0, sim_1)
            LL_nhpp_b = _score_nhpp_loglik(nhpp_b, sim_0, sim_1)
            LL_hawkes_b = float(hawkes_b["loglik_in_sample"])
            bootstrap_LR[k] = 2.0 * (LL_hawkes_b - LL_nhpp_b)
        except Exception:
            n_failed += 1
            bootstrap_LR[k] = np.nan

    valid = bootstrap_LR[~np.isnan(bootstrap_LR)]
    p_value = float(np.mean(valid >= LR_observed)) if valid.size > 0 else float("nan")

    # 3. Diagnostic plot (SC-3): histogram must visibly show point mass at 0 + continuous tail
    if diagnostic_plot_path is not None:
        Path(diagnostic_plot_path).parent.mkdir(parents=True, exist_ok=True)
        fig, ax = plt.subplots(figsize=(7, 4))
        # Use log scale on y-axis to make point mass at 0 visible alongside continuous tail
        ax.hist(valid, bins=60, color="#3a6ea5", edgecolor="black", alpha=0.85)
        ax.axvline(LR_observed, color="red", linestyle="--", linewidth=1.5,
                   label=f"observed LR = {LR_observed:.3f}")
        ax.set_yscale("log")
        ax.set_xlabel("Bootstrap LR statistic (2 * ΔLL)")
        ax.set_ylabel("Frequency (log scale)")
        ax.set_title(f"Bootstrap LR null distribution (n_reps={n_reps}, n_failed={n_failed})\n"
                     f"χ²(0) point mass at 0 + χ²(1) continuous tail")
        ax.legend(loc="upper right")
        fig.tight_layout()
        fig.savefig(diagnostic_plot_path, dpi=120)
        plt.close(fig)

    return {
        "observed_stat": float(LR_observed),
        "bootstrap_null_dist_50_50_chi2_0_chi2_1": valid.tolist(),
        "p_value": p_value,
        "rejects_at_alpha": bool(p_value < alpha) if not np.isnan(p_value) else False,
        "n_reps": int(n_reps),
        "n_failed": int(n_failed),
        "seed": int(seed),
        "alpha": float(alpha),
    }
```

"""DGP-03: parametric bootstrap LR test."""
import subprocess
from pathlib import Path

import numpy as np
import pytest

ANALYSIS_ROOT = Path(__file__).resolve().parents[1]
WINDOW_START = 0.0
WINDOW_END = 2_592_000.0  # 30 days


def test_null_distribution_mixture_shape(synthetic_nhpp_baseline_only_legs):
    leg_0, leg_1 = synthetic_nhpp_baseline_only_legs
    result = parametric_bootstrap_lr(
        leg_0, leg_1, panel_data_hash="test-nhpp-mixture-shape",
        window_start=WINDOW_START, window_end=WINDOW_END, n_reps=200,
    )
    null = np.asarray(result["bootstrap_null_dist_50_50_chi2_0_chi2_1"])
    assert null.size > 0, "Bootstrap null distribution is empty"
    near_zero_frac = float(np.mean(null <= 1e-6))
    continuous_frac = float(np.mean(null > 1e-3))
    # Mixture signature: NOT all-zero AND NOT continuous-only
    assert near_zero_frac < 1.0, f"All values at zero — bootstrap is broken (continuous tail missing)"
    assert continuous_frac < 1.0, f"All values continuous — bootstrap is broken (point mass at 0 missing)"


def test_power_on_synthetic_hawkes(synthetic_hawkes_eta_05_legs):
    leg_0, leg_1 = synthetic_hawkes_eta_05_legs
    result = parametric_bootstrap_lr(
        leg_0, leg_1, panel_data_hash="test-hawkes-power",
