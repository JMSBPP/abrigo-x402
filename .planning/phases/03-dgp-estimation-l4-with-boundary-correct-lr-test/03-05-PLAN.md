"""DGP-05: Brown et al. 2002 time-rescaling KS test on held-out Hawkes residuals.

LOCKED INVARIANTS (PRE_REGISTRATION + PITFALLS §4):
- Compensator Λ_i(t) for exponential-kernel Hawkes computed in CLOSED FORM (no numerical integration).
- KS test applied on HELD-OUT segment with TRAIN-FITTED parameters (Pitfall 5: in-sample optimism).
- Acceptance threshold: p > 0.05 (PRE_REGISTRATION §Test Statistics).
- scipy.stats.kstest(rescaled_dt, 'expon') is the test statistic.
"""
from __future__ import annotations

import numpy as np
import polars as pl
from scipy.stats import kstest


def compute_compensator_exp_kernel(
    event_times: np.ndarray,
    leg_idx: int,
    baseline: np.ndarray,         # shape (2,)
    adjacency: np.ndarray,        # shape (2, 2)
    decays: float,
    full_history_leg_0: np.ndarray,
    full_history_leg_1: np.ndarray,
    window_start: float,
) -> np.ndarray:
    """Closed-form Λ_i(t) - Λ_i(window_start) for each event_time in the held-out window.

    For each prior event t_jk (in train OR earlier held-out):
        contribution = α_ij * (exp(-β * max(W_start - t_jk, 0)) - exp(-β * (t - t_jk))) / β
        (the integral of α_ij exp(-β (s - t_jk)) over s ∈ [max(W_start, t_jk), t])
    The baseline term is μ_i * (t - W_start).
    """
    baseline = np.asarray(baseline, dtype=np.float64)
    adjacency = np.asarray(adjacency, dtype=np.float64)
    mu_i = float(baseline[leg_idx])
    beta = float(decays)
    Lambda_at_events = np.empty(event_times.size, dtype=np.float64)
    histories = (np.sort(full_history_leg_0).astype(np.float64),
                 np.sort(full_history_leg_1).astype(np.float64))
    for idx, t in enumerate(event_times):
        L_baseline = mu_i * (float(t) - float(window_start))
        L_exc = 0.0
        for j, hist in enumerate(histories):
            # events strictly before t contribute (note: t itself excluded for the integral lower bound)
            past = hist[hist < t]
            if past.size == 0:
                continue
            lower = np.maximum(window_start - past, 0.0)
            term = (np.exp(-beta * lower) - np.exp(-beta * (t - past))) / beta
            L_exc += float(adjacency[leg_idx, j]) * float(np.sum(term))
        Lambda_at_events[idx] = L_baseline + L_exc
    return Lambda_at_events


def time_rescaling_ks_test_leg(
    held_out_event_times: np.ndarray,
    leg_idx: int,
    baseline: np.ndarray,
    adjacency: np.ndarray,
    decays: float,
    full_history_leg_0: np.ndarray,
    full_history_leg_1: np.ndarray,
    window_start: float,
    window_end: float,
) -> dict:
    """KS test on rescaled inter-arrival times under exponential-kernel Hawkes.

    Returns dict with keys: ks_statistic, p_value, n_events, rescaled_dt, Lambda_at_events.
    """
    if held_out_event_times.size < 2:
        return {
            "ks_statistic": float("nan"),
            "p_value": float("nan"),
            "n_events": int(held_out_event_times.size),
            "rescaled_dt": np.empty(0, dtype=np.float64).tolist(),
            "Lambda_at_events": np.empty(0, dtype=np.float64).tolist(),
            "skipped_reason": "insufficient_events",
        }

    Lambda = compute_compensator_exp_kernel(
        held_out_event_times, leg_idx, baseline, adjacency, decays,
        full_history_leg_0, full_history_leg_1, window_start,
    )

    rescaled_dt = np.diff(np.concatenate([[0.0], Lambda]))
    # Drop any non-positive rescaled_dt (numerical edge cases)
    rescaled_dt = rescaled_dt[rescaled_dt > 0.0]

    if rescaled_dt.size < 2:
        return {
            "ks_statistic": float("nan"),
            "p_value": float("nan"),
            "n_events": int(held_out_event_times.size),
            "rescaled_dt": rescaled_dt.tolist(),
            "Lambda_at_events": Lambda.tolist(),
            "skipped_reason": "non_positive_rescaling",
        }

    ks_stat, ks_pvalue = kstest(rescaled_dt, "expon")
    return {
        "ks_statistic": float(ks_stat),
        "p_value": float(ks_pvalue),
        "n_events": int(held_out_event_times.size),
        "rescaled_dt": rescaled_dt.tolist(),
        "Lambda_at_events": Lambda.tolist(),
    }


def build_residuals_dataframe(
    per_leg_results: list[dict],
    held_out_event_times_per_leg: list[np.ndarray],
) -> pl.DataFrame:
    """Combine per-leg time-rescaling outputs into a single DataFrame for residuals.parquet.

    Columns: leg (UInt8), event_time (Float64), Lambda_at_event (Float64), rescaled_dt (Float64).
    """
    rows: list[dict] = []
    for leg_idx, (result, event_times) in enumerate(zip(per_leg_results, held_out_event_times_per_leg)):
        Lambdas = np.asarray(result.get("Lambda_at_events", []), dtype=np.float64)
        rescaled = np.asarray(result.get("rescaled_dt", []), dtype=np.float64)
        # rescaled_dt is one shorter than events for the leading 0-anchor; pad with NaN at index 0
        rescaled_aligned = np.full(event_times.size, float("nan"), dtype=np.float64)
        n = min(event_times.size, rescaled.size)
        rescaled_aligned[:n] = rescaled[:n]
        for t, Lam, rdt in zip(event_times, Lambdas, rescaled_aligned):
            rows.append({
                "leg": int(leg_idx),
                "event_time": float(t),
                "Lambda_at_event": float(Lam),
                "rescaled_dt": float(rdt),
            })
    if not rows:
        return pl.DataFrame(schema={
            "leg": pl.UInt8, "event_time": pl.Float64,
            "Lambda_at_event": pl.Float64, "rescaled_dt": pl.Float64,
        })
    return pl.DataFrame(rows).with_columns(pl.col("leg").cast(pl.UInt8))

"""DGP-05: time-rescaling KS test on held-out Hawkes residuals."""
import numpy as np
import polars as pl

from abrigo_x402.dgp.time_rescaling import (
    compute_compensator_exp_kernel, time_rescaling_ks_test_leg, build_residuals_dataframe,
)


def test_compensator_closed_form():
    """Manual hand-computation: μ=0.1, α=0.5, β=1.0, one prior event at t=0.5 (in train, W_start=1.0),
    one held-out event at t=2.0. Λ_0(2.0) - Λ_0(W_start=1.0) should match analytic formula.
    """
    baseline = np.array([0.1, 0.0], dtype=np.float64)
    adjacency = np.array([[0.5, 0.0], [0.0, 0.0]], dtype=np.float64)
    decays = 1.0
    # Prior event in train at t_jk = 0.5; held-out event at t = 2.0
    full_history_leg_0 = np.array([0.5, 2.0], dtype=np.float64)
    full_history_leg_1 = np.array([], dtype=np.float64)
    held_out_times = np.array([2.0], dtype=np.float64)
    W_start = 1.0
    Lambda = compute_compensator_exp_kernel(
        held_out_times, 0, baseline, adjacency, decays,
        full_history_leg_0, full_history_leg_1, W_start,
    )
    # Expected: μ*(t-W_start) + α * (exp(-β*max(W_start - 0.5, 0)) - exp(-β*(2.0 - 0.5))) / β
    #         = 0.1*(2.0-1.0) + 0.5 * (exp(-0.5) - exp(-1.5)) / 1.0
    expected = 0.1 * 1.0 + 0.5 * (np.exp(-0.5) - np.exp(-1.5)) / 1.0
    assert abs(float(Lambda[0]) - float(expected)) < 1e-9, (
        f"Compensator closed-form mismatch: got {Lambda[0]}, expected {expected}"
    )


def test_passes_on_true_model(synthetic_hawkes_eta_05_legs):
    """KS test on held-out with TRUE parameters should not reject."""
    leg_0, leg_1 = synthetic_hawkes_eta_05_legs
    W_start, W_end = 0.0, 2_592_000.0
    t_split = W_start + 0.80 * (W_end - W_start)
    held_0 = leg_0[leg_0 >= t_split]
    held_1 = leg_1[leg_1 >= t_split]
    # Use the locked synthetic parameters from conftest fixture (η=0.5 panel was generated with these)
    # We approximate by using the panel's empirical baseline ≈ events/sec on train
    train_dur = t_split - W_start
    baseline = np.array([
        float(leg_0[leg_0 < t_split].size) / train_dur,
        float(leg_1[leg_1 < t_split].size) / train_dur,
    ])
    # With self-excitation η=0.5 and symmetric adjacency: α ≈ 0.5 * β / 2 per off-diagonal
    decays = 0.1
    adjacency = np.full((2, 2), 0.5 * decays / 2.0, dtype=np.float64)
    results = [
        time_rescaling_ks_test_leg(held_0, 0, baseline, adjacency, decays, leg_0, leg_1, t_split, W_end),
        time_rescaling_ks_test_leg(held_1, 1, baseline, adjacency, decays, leg_0, leg_1, t_split, W_end),
    ]
    # At least one leg should have p > 0.05 (this is a loose check — synthetic generator approximation)
    passes = [r["p_value"] > 0.05 for r in results if not np.isnan(r["p_value"])]
    assert any(passes) or all(np.isnan(r["p_value"]) for r in results), (
        f"Both legs reject under true-parameter rescaling: {results}"
    )


def test_fails_on_misspecified(synthetic_hawkes_eta_05_legs):
    """KS test on held-out under NHPP-only (α=0) rescaling should reject the true-Hawkes data."""
    leg_0, leg_1 = synthetic_hawkes_eta_05_legs
    W_start, W_end = 0.0, 2_592_000.0
    t_split = W_start + 0.80 * (W_end - W_start)
    held_0 = leg_0[leg_0 >= t_split]
    held_1 = leg_1[leg_1 >= t_split]
    train_dur = t_split - W_start
    baseline = np.array([
        float(leg_0[leg_0 < t_split].size) / train_dur,
        float(leg_1[leg_1 < t_split].size) / train_dur,
    ])
    # Misspecified: α=0 (pure NHPP rescaling of true Hawkes data)
    adjacency = np.zeros((2, 2), dtype=np.float64)
    result = time_rescaling_ks_test_leg(held_0, 0, baseline, adjacency, 0.1, leg_0, leg_1, t_split, W_end)
    # If sample is too small the test is skipped; otherwise we expect rejection on at least one leg
    if not np.isnan(result["p_value"]):
        # Allow either leg to reject; sometimes the leg with less self-excitation appears NHPP-like
        result_1 = time_rescaling_ks_test_leg(held_1, 1, baseline, adjacency, 0.1, leg_0, leg_1, t_split, W_end)
        p_values = [p for p in (result["p_value"], result_1["p_value"]) if not np.isnan(p)]
        assert any(p < 0.05 for p in p_values), (
            f"Misspecified NHPP rescaling failed to reject Hawkes data: p_values={p_values}"
        )


def test_residuals_dataframe_schema():
    held_out_times = np.array([100.0, 200.0, 300.0], dtype=np.float64)
    per_leg = [
        {
            "Lambda_at_events": [1.0, 2.0, 3.0],
            "rescaled_dt": [1.0, 1.0, 1.0],
        },
        {
            "Lambda_at_events": [],
            "rescaled_dt": [],
        },
    ]
    df = build_residuals_dataframe(per_leg, [held_out_times, np.empty(0, dtype=np.float64)])
    assert df.columns == ["leg", "event_time", "Lambda_at_event", "rescaled_dt"]
    assert df.schema["leg"] == pl.UInt8
    assert df.schema["event_time"] == pl.Float64
    assert df.schema["Lambda_at_event"] == pl.Float64
    assert df.schema["rescaled_dt"] == pl.Float64
