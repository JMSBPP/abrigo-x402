```python
"""DGP orchestrator: load panel, split, fit NHPP+Hawkes, bootstrap LR, held-out LL, KS rescaled-time, profile-likelihood CI, assemble fit_report.json.

LOCKED INVARIANTS:
- run_id = sha256(panel_dataHash + fit_code_gitCommit + tick_lib_version)[:12] — CONTEXT.md decision.
- fit_report.json carries PANEL-02 metadata header (chainId, contractAddress, blockRange, fetchTimestamp, dataHash, gitCommit).
- ALL SC-1 keys present even when gate fails (CONTEXT.md <specifics>: NEVER write a fit_report.json with missing keys).
- residuals.parquet sidecar; residuals_hash = sha256(file_bytes).
"""
from __future__ import annotations

import hashlib
import json
import subprocess
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import numpy as np
import polars as pl
import tick

from abrigo_x402.dgp.held_out import (
    HELD_OUT_FRACTION_DEFAULT, wall_clock_split,
    compute_held_out_loglik_hawkes, compute_held_out_loglik_nhpp,
)
from abrigo_x402.dgp.hawkes_fit import fit_hawkes_expkern
from abrigo_x402.dgp.lr_test import parametric_bootstrap_lr, PRODUCTION_N_REPS
from abrigo_x402.dgp.nhpp_inar import fit_nhpp_inar
from abrigo_x402.dgp.profile_likelihood import profile_likelihood_eta_ci
from abrigo_x402.dgp.stationarity import baseline_stationarity_check
from abrigo_x402.dgp.time_rescaling import time_rescaling_ks_test_leg, build_residuals_dataframe


REQUIRED_FIT_REPORT_KEYS: tuple[str, ...] = (
    "chainId", "contractAddress", "blockRange", "fetchTimestamp", "dataHash", "gitCommit",
    "run_id", "tick_lib_version",
    "nhpp_inar_params", "hawkes_mv_params",
    "lr_test", "ks_rescaled_time", "held_out_loglik",
    "branching_ratio_ci", "baseline_stationarity_check",
    "input_diagnostics", "gate_passes", "gate_criteria",
)

ETA_FLOOR: float = 0.2
LR_ALPHA: float = 0.01
KS_ALPHA: float = 0.05


def _git_commit() -> str:
    try:
        return subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip()
    except Exception:
        return "unknown"


def _derive_run_id(panel_data_hash: str, git_commit: str, tick_version: str) -> str:
    h = hashlib.sha256((panel_data_hash + git_commit + tick_version).encode()).hexdigest()
    return h[:12]


def _extract_legs_from_panel(panel_df: pl.DataFrame) -> tuple[np.ndarray, np.ndarray, float, float]:
    """leg_0 = token0-inflow Swaps; leg_1 = token1-inflow Swaps (CONTEXT.md decision)."""
    swaps = panel_df.filter(pl.col("event_name") == "Swap")
    if "block_timestamp" not in swaps.columns:
        raise KeyError("panel parquet lacks 'block_timestamp' column required for DGP fit")
    amount0_int = swaps.get_column("amount0").cast(pl.Int128)
    amount1_int = swaps.get_column("amount1").cast(pl.Int128)
    ts = swaps.get_column("block_timestamp").cast(pl.Float64).to_numpy()
    leg_0_mask = amount0_int.gt(0).to_numpy()
    leg_1_mask = amount1_int.gt(0).to_numpy()
    leg_0 = np.sort(ts[leg_0_mask])
    leg_1 = np.sort(ts[leg_1_mask])
    if leg_0.size == 0 and leg_1.size == 0:
        raise ValueError("panel produced zero events on both legs")
    window_start = float(ts.min())
    window_end = float(ts.max())
    return leg_0, leg_1, window_start, window_end


def _compute_data_hash(panel_path: Path) -> str:
    return hashlib.sha256(panel_path.read_bytes()).hexdigest()


def _panel_provenance(panel_df: pl.DataFrame, panel_path: Path) -> dict[str, Any]:
    chain_id = 42220
    contract_address = "0x61Ef8708fc240DC7f9F2c0d81c3124Df2fd8829F"
    stem = panel_path.stem
    try:
        from_block, to_block = stem.split("_", 1)
        block_range = [int(from_block), int(to_block)]
    except Exception:
        block_range = [0, 0]
    return {
        "chainId": chain_id,
        "contractAddress": contract_address,
        "blockRange": block_range,
        "fetchTimestamp": datetime.now(timezone.utc).isoformat(),
        "dataHash": _compute_data_hash(panel_path),
        "gitCommit": _git_commit(),
    }


@dataclass(frozen=True)
class FitOutput:
    run_id: str
    fit_report_path: Path
    residuals_path: Path
    fit_report: dict[str, Any]


def run_fit(
    panel_path: str | Path,
    out_dir: str | Path,
    bootstrap_reps: int = PRODUCTION_N_REPS,
    decays: float = 0.1,
) -> FitOutput:
    panel_path = Path(panel_path)
    panel_df = pl.read_parquet(panel_path)
    provenance = _panel_provenance(panel_df, panel_path)

    leg_0, leg_1, window_start, window_end = _extract_legs_from_panel(panel_df)

    run_id = _derive_run_id(provenance["dataHash"], provenance["gitCommit"], tick.__version__)
    run_dir = Path(out_dir) / run_id
    run_dir.mkdir(parents=True, exist_ok=True)

    split = wall_clock_split(leg_0, leg_1, window_start, window_end, HELD_OUT_FRACTION_DEFAULT)

    nhpp_train = fit_nhpp_inar(split.train_leg_0, split.train_leg_1, window_start, split.t_split)
    hawkes_train = fit_hawkes_expkern(split.train_leg_0, split.train_leg_1, decays=decays)

    lr_result = parametric_bootstrap_lr(
        leg_0, leg_1, panel_data_hash=provenance["dataHash"],
        window_start=window_start, window_end=window_end,
        n_reps=bootstrap_reps, alpha=LR_ALPHA,
        diagnostic_plot_path=str(Path("reports/_diagnostics/lr_null_dist.png")),
    )

    nhpp_baseline_per_sec = np.asarray(nhpp_train["intercept"], dtype=np.float64) / float(nhpp_train["bin_width_seconds"])
    nhpp_held_out_ll = compute_held_out_loglik_nhpp(
        nhpp_baseline_per_sec, split.held_out_leg_0, split.held_out_leg_1, split.t_split, window_end,
    )
    hawkes_held_out_ll = compute_held_out_loglik_hawkes(
        np.asarray(hawkes_train["baseline"]),
        np.asarray(hawkes_train["adjacency"]),
        float(hawkes_train["decays"]),
        split.held_out_leg_0, split.held_out_leg_1,
        leg_0, leg_1,
        split.t_split, window_end,
    )

    stationarity = baseline_stationarity_check(split)

    ks_per_leg = []
    for leg_idx, held_times in enumerate((split.held_out_leg_0, split.held_out_leg_1)):
        result = time_rescaling_ks_test_leg(
            held_times, leg_idx,
            np.asarray(hawkes_train["baseline"]),
            np.asarray(hawkes_train["adjacency"]),
            float(hawkes_train["decays"]),
            leg_0, leg_1,
            split.t_split, window_end,
        )
        ks_per_leg.append(result)

    ks_pvals = [r["p_value"] for r in ks_per_leg if not np.isnan(r["p_value"])]
    ks_combined_pvalue = float(min(ks_pvals)) if ks_pvals else float("nan")

    eta_ci = profile_likelihood_eta_ci(leg_0, leg_1, hawkes_train, decays=decays, alpha=0.05)

    residuals_df = build_residuals_dataframe(ks_per_leg, [split.held_out_leg_0, split.held_out_leg_1])
    residuals_path = run_dir / "residuals.parquet"
    residuals_df.write_parquet(residuals_path)
    residuals_hash = hashlib.sha256(residuals_path.read_bytes()).hexdigest()

    eta_hat = float(hawkes_train["branching_ratio"])
    lr_rejects = bool(lr_result.get("rejects_at_alpha", False))
    eta_floor_met = bool(eta_hat >= ETA_FLOOR)
    ks_held_out_passes = bool(not np.isnan(ks_combined_pvalue) and ks_combined_pvalue > KS_ALPHA)
    branching_ci_excludes_zero = bool(eta_ci["lower"] > 0.0)
    stationary = bool(stationarity["decision"] == "stationary")
    gate_passes = bool(lr_rejects and eta_floor_met and ks_held_out_passes and branching_ci_excludes_zero and stationary)

    fit_report: dict[str, Any] = {
        **provenance,
        "run_id": run_id,
        "tick_lib_version": str(tick.__version__),
        "nhpp_inar_params": nhpp_train,
        "hawkes_mv_params": hawkes_train,
        "lr_test": lr_result,
        "ks_rescaled_time": {
            "p_value": ks_combined_pvalue,
            "residuals_hash": residuals_hash,
            "per_leg": [
                {k: r[k] for k in ("ks_statistic", "p_value", "n_events") if k in r}
                for r in ks_per_leg
            ],
        },
        "held_out_loglik": {
            "nhpp": float(nhpp_held_out_ll),
            "hawkes": float(hawkes_held_out_ll),
            "split_metadata": split.to_metadata(),
        },
        "branching_ratio_ci": eta_ci,
        "baseline_stationarity_check": stationarity,
        "input_diagnostics": {
            "total_events_per_leg": [int(leg_0.size), int(leg_1.size)],
            "simultaneous_event_pairs": int(np.sum(np.isin(leg_0, leg_1))),
            "tie_counts": {
                "block_level_ties_leg_0": int(np.sum(np.diff(np.sort(leg_0)) == 0.0)),
                "block_level_ties_leg_1": int(np.sum(np.diff(np.sort(leg_1)) == 0.0)),
            },
        },
        "gate_passes": gate_passes,
        "gate_criteria": {
            "lr_rejects": lr_rejects,
            "eta_floor_met": eta_floor_met,
            "ks_held_out_passes": ks_held_out_passes,
            "branching_ci_excludes_zero": branching_ci_excludes_zero,
            "stationary": stationary,
        },
    }

    missing = [k for k in REQUIRED_FIT_REPORT_KEYS if k not in fit_report]
    if missing:
        raise KeyError(f"fit_report.json missing required SC-1 keys: {missing}")

    fit_report_path = run_dir / "fit_report.json"
    fit_report_path.write_text(json.dumps(fit_report, indent=2, sort_keys=True))

    return FitOutput(
        run_id=run_id,
        fit_report_path=fit_report_path,
        residuals_path=residuals_path,
        fit_report=fit_report,
    )
```
