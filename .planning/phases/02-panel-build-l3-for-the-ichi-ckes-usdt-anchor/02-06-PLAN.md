```typescript
import { type Address, parseAbi, type PublicClient } from 'viem';
import { celoClient } from '../viem-clients.js';
import { appendLedger } from '../cost-ledger.js';
import { writeFileSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';

export const MENTO_ADDRESSES = {
  Broker: '0x777A8255cA72412f0d706dc03C9D1987306B4CaD',
  BiPoolManager: '0x22d9db95E6Ae61c104A7B6F6C78D7993B94ec901',
  cKES: '0x456a3D042C0DbD3db53D5489e98dFb038553B0d0',
  USDm: '0x765DE816845861e75A25fCA122bb6898B8B1282a',
} as const satisfies Record<string, Address>;

export const BROKER_ABI = parseAbi([
  'function getAmountOut(address exchangeProvider, bytes32 exchangeId, address tokenIn, address tokenOut, uint256 amountIn) view returns (uint256)',
]);

export interface FxSnapRow {
  block: number;
  source: 'mento-broker';
  rate_x1e18: string;
  method: 'exact' | 'unavailable';
  provenance_url: string;
}

export interface FxSnapOptions {
  blocks: number[];
  exchangeId: `0x${string}`;
  tokenIn?: Address;          // default cKES
  tokenOut?: Address;         // default USDm
  amountIn?: bigint;          // default 1e18 (1 unit)
  client?: PublicClient;      // default celoClient
  outputPath: string;
}

export async function snapFxBlocks(opts: FxSnapOptions): Promise<FxSnapRow[]> {
  const client = (opts.client ?? celoClient) as PublicClient;
  const tokenIn = opts.tokenIn ?? MENTO_ADDRESSES.cKES;
  const tokenOut = opts.tokenOut ?? MENTO_ADDRESSES.USDm;
  const amountIn = opts.amountIn ?? (10n ** 18n);
  const dedupedBlocks = Array.from(new Set(opts.blocks)).sort((a, b) => a - b);

  const rows: FxSnapRow[] = [];
  for (const block of dedupedBlocks) {
    let row: FxSnapRow;
    try {
      const out = await client.readContract({
        address: MENTO_ADDRESSES.Broker as Address,
        abi: BROKER_ABI,
        functionName: 'getAmountOut',
        args: [
          MENTO_ADDRESSES.BiPoolManager as Address,
          opts.exchangeId,
          tokenIn,
          tokenOut,
          amountIn,
        ],
        blockNumber: BigInt(block),  // LOAD-BEARING — RESEARCH §B + Pitfall 1
      });
      row = {
        block,
        source: 'mento-broker',
        rate_x1e18: (out as bigint).toString(),
        method: 'exact',
        provenance_url: `https://celo.blockscout.com/block/${block}`,
      };
    } catch {
      row = {
        block,
        source: 'mento-broker',
        rate_x1e18: '0',
        method: 'unavailable',
        provenance_url: `https://celo.blockscout.com/block/${block}`,
      };
    }
    rows.push(row);

    try {
      await appendLedger({
        ts: new Date().toISOString(),
        endpoint: 'forno',
        chain: 'celo',
        usdc_cost: '0',
        paid_real: false,
        blocks_queried: 1,
        request_id: `mento-fx-${block}`,
      });
    } catch { /* best-effort in tests */ }
  }

  mkdirSync(dirname(opts.outputPath), { recursive: true });
  const lines = rows.map(r => JSON.stringify(r)).join('\n');
  writeFileSync(opts.outputPath, lines + '\n');

  return rows;
}
```

```python
"""PANEL-03: Read Mento TS sidecar JSONL; join panel events to FX rates; forward-fill;
USDT/USD as SEPARATE column never collapsed to 1.0 (CLAUDE.md non-negotiable)."""
from pathlib import Path
import json
import polars as pl

FX_METHODS = ("exact", "forward_fill", "unavailable")

FX_SIDECAR_SCHEMA = {
    "block": pl.Int64,
    "source": pl.String,
    "rate_x1e18": pl.String,
    "method": pl.String,
    "provenance_url": pl.String,
}

def load_fx_sidecar(sidecar_path: Path) -> pl.DataFrame:
    """Read TS-sidecar JSONL of Mento broker historical-block FX rates."""
    sidecar_path = Path(sidecar_path)
    rows = []
    with open(sidecar_path, "r") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rows.append(json.loads(line))
    return pl.DataFrame(rows, schema=FX_SIDECAR_SCHEMA)

def attach_rates(panel_df: pl.DataFrame, fx_df: pl.DataFrame) -> pl.DataFrame:
    """For each panel row, find the most recent FX rate at block <= event_block (asof join).

    Adds columns:
      cKES_per_USDm_rate (String; decimal-string from uint256 1e18-scaled)
      fx_method ("exact" | "forward_fill" | "unavailable")
      fx_provenance_url (Blockscout block URL)
      usdt_usd_rate (String; default "1.0" — NEVER absent — CLAUDE.md non-negotiable)
      usdt_usd_method ("stipulated" — Phase 4 may overwrite to "oracle" or "depeg-event")

    Forward-fill semantics:
      - Sidecar rows with method='unavailable' are EXCLUDED before the asof join,
        so forward-fill takes the most recent 'exact' rate.
      - Panel event at block < any exact-rate block → fx_method='unavailable'.
    """
    if "blockNumber" not in panel_df.columns:
        raise ValueError("panel_df must have blockNumber column")

    # Filter sidecar to exact rates only (forward-fill skips unavailable rows)
    exact_fx = fx_df.filter(pl.col("method") == "exact").sort("block")

    panel_sorted = panel_df.sort("blockNumber")

    # asof join: each panel row gets the most recent exact-rate block <= panel.blockNumber
    if exact_fx.height == 0:
        out = panel_sorted.with_columns(
            pl.lit(None, dtype=pl.String).alias("cKES_per_USDm_rate"),
            pl.lit("unavailable").alias("fx_method"),
            pl.lit(None, dtype=pl.String).alias("fx_provenance_url"),
        )
    else:
        joined = panel_sorted.join_asof(
            exact_fx.rename({"block": "_fx_block", "rate_x1e18": "cKES_per_USDm_rate",
                             "provenance_url": "fx_provenance_url"}),
            left_on="blockNumber",
            right_on="_fx_block",
            strategy="backward",  # most recent block ≤ event block
        )
        out = joined.with_columns(
            # fx_method = 'exact' if event_block == fx_block, else 'forward_fill', else 'unavailable'
            pl.when(pl.col("cKES_per_USDm_rate").is_null())
              .then(pl.lit("unavailable"))
              .when(pl.col("_fx_block") == pl.col("blockNumber"))
              .then(pl.lit("exact"))
              .otherwise(pl.lit("forward_fill"))
              .alias("fx_method"),
        ).drop(["_fx_block", "source", "method"], strict=False)

    # USDT/USD column — ALWAYS present, never collapsed (CLAUDE.md non-negotiable)
    out = out.with_columns(
        pl.lit("1.0").alias("usdt_usd_rate"),
        pl.lit("stipulated").alias("usdt_usd_method"),
    )
    return out
```
