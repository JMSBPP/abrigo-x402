```typescript
import { type Address, parseAbi, type PublicClient } from 'viem';
import { celoClient } from '../viem-clients.js';
import { appendLedger } from '../cost-ledger.js';
import { writeFileSync, mkdirSync } from 'node:fs';
import { dirname } from 'node:path';

export const VAULT_ADDRESSES = {
  cKES_USDT_anchor: '0xe304b980535c29869983BC58d129F984Fec4176F',
  cKES_USDT_pool: '0x61Ef8708fc240DC7f9F2c0d81c3124Df2fd8829F',
} as const satisfies Record<string, Address>;

export const MULTICALL3_CELO: Address = '0xcA11bde05977b3631167028862bE2a173976CA11';

// Verified against analysis/tests/fixtures/ichi_vault_abi.json (Plan 02-00 capture).
// If the fixture pins different tick-getter names, REPLACE these signatures and
// document the drift in this plan's SUMMARY.
export const VAULT_ABI = parseAbi([
  'function totalAmounts() view returns (uint256 total0, uint256 total1)',
  'function totalSupply() view returns (uint256)',
  'function currentTick() view returns (int24)',
  'function baseLower() view returns (int24)',
  'function baseUpper() view returns (int24)',
]);

export interface VaultStateRow {
  blockNumber: number;
  totalAmounts_0: string | null;
  totalAmounts_1: string | null;
  totalSupply: string | null;
  currentTick: number | null;
  lowerTick: number | null;
  upperTick: number | null;
}

export interface VaultStateOptions {
  vaultAddress: Address;
  blocks: number[];                       // may contain duplicates; per-block memo deduplicates
  client?: PublicClient;                  // default celoClient (Forno)
  outputPath: string;                     // JSONL output path
}

export async function snapVaultState(opts: VaultStateOptions): Promise<VaultStateRow[]> {
  const client = (opts.client ?? celoClient) as PublicClient;
  const dedupedBlocks = Array.from(new Set(opts.blocks)).sort((a, b) => a - b);
  const rows: VaultStateRow[] = [];

  for (const block of dedupedBlocks) {
    const results = await client.multicall({
      multicallAddress: MULTICALL3_CELO,
      blockNumber: BigInt(block),
      allowFailure: true,
      contracts: [
        { address: opts.vaultAddress, abi: VAULT_ABI, functionName: 'totalAmounts' },
        { address: opts.vaultAddress, abi: VAULT_ABI, functionName: 'totalSupply' },
        { address: opts.vaultAddress, abi: VAULT_ABI, functionName: 'currentTick' },
        { address: opts.vaultAddress, abi: VAULT_ABI, functionName: 'baseLower' },
        { address: opts.vaultAddress, abi: VAULT_ABI, functionName: 'baseUpper' },
      ],
    });

    const [ta, ts, ct, lt, ut] = results as any[];
    rows.push({
      blockNumber: block,
      totalAmounts_0: ta?.status === 'success' ? (ta.result as [bigint, bigint])[0].toString() : null,
      totalAmounts_1: ta?.status === 'success' ? (ta.result as [bigint, bigint])[1].toString() : null,
      totalSupply: ts?.status === 'success' ? (ts.result as bigint).toString() : null,
      currentTick: ct?.status === 'success' ? Number(ct.result) : null,
      lowerTick: lt?.status === 'success' ? Number(lt.result) : null,
      upperTick: ut?.status === 'success' ? Number(ut.result) : null,
    });

    // DEMAND-01 enforce: every Forno call → endpoint='forno' ledger row (uncapped)
    try {
      await appendLedger({
        ts: new Date().toISOString(),
        endpoint: 'forno',
        chain: 'celo',
        usdc_cost: '0',
        paid_real: false,
        blocks_queried: 1,
        request_id: `vault-state-${opts.vaultAddress}-${block}`,
      });
    } catch { /* ledger best-effort in tests; tolerate missing dir */ }
  }

  // Write JSONL sidecar
  mkdirSync(dirname(opts.outputPath), { recursive: true });
  const lines = rows.map(r => JSON.stringify(r)).join('\n');
  writeFileSync(opts.outputPath, lines + '\n');

  return rows;
}
```

`analysis/src/abrigo_x402/vault_state.py`:
```python
"""PANEL-01 vault-state component: read TS sidecar JSONL, compute vault_in_range per Swap.

The TS sidecar (`fetch/src/vault/state-snap.ts`) writes JSONL with one row per
DEDUPED block in the panel's Swap-event range. This module reads it as polars,
left-joins on `blockNumber`, and computes `vault_in_range = lowerTick <= tick <= upperTick`.

Conservative semantics: if a Swap has no matching vault-state block (left-join NULL),
`vault_in_range` is False — the vault accrues zero LP-fees for that Swap. This is
intentional: missing vault state is treated as "no info" → no fee credit.
"""
from pathlib import Path
import json
import polars as pl

VAULT_STATE_SCHEMA = {
    "blockNumber": pl.Int64,
    "totalAmounts_0": pl.String,
    "totalAmounts_1": pl.String,
    "totalSupply": pl.String,
    "currentTick": pl.Int32,
    "lowerTick": pl.Int32,
    "upperTick": pl.Int32,
}

def load_vault_state(sidecar_path: Path) -> pl.DataFrame:
    """Read the TS-sidecar-generated JSONL of vault state per block."""
    sidecar_path = Path(sidecar_path)
    rows = []
    with open(sidecar_path, "r") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            rows.append(json.loads(line))
    return pl.DataFrame(rows, schema=VAULT_STATE_SCHEMA)

def attach_in_range(swap_df: pl.DataFrame, vault_df: pl.DataFrame) -> pl.DataFrame:
    """Left-join vault_df onto swap_df by blockNumber; compute vault_in_range flag.

    Adds columns from vault state: totalAmounts_0, totalAmounts_1, totalSupply,
    currentTick (vault-side, NOT renamed — swap-side `tick` is the in-range comparison anchor),
    lowerTick, upperTick, vault_in_range.

    vault_in_range = (lowerTick <= swap.tick <= upperTick) when vault state exists;
    False when vault state is missing for the swap's block.
    """
    if "blockNumber" not in swap_df.columns:
        raise ValueError("swap_df must have blockNumber column for join")
    # Rename vault's currentTick to avoid collision with swap's tick column
    vd = vault_df.rename({"currentTick": "vault_currentTick"})
    out = swap_df.join(vd, on="blockNumber", how="left")
    # vault_in_range: lowerTick <= swap.tick <= upperTick; null lowerTick/upperTick → False
    out = out.with_columns(
        (
            pl.col("lowerTick").is_not_null() &
            pl.col("upperTick").is_not_null() &
            (pl.col("lowerTick") <= pl.col("tick")) &
            (pl.col("tick") <= pl.col("upperTick"))
        ).alias("vault_in_range")
    )
    return out
```
