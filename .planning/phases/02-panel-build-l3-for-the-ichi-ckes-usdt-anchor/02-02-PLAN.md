```python
"""PANEL-01: ABI-decoders for Uniswap V3 Swap/Mint/Burn + ICHI vault Deposit/Withdraw.

Python-side decode using eth_abi against the raw `topics` + `data` columns
emitted by ingest.load_jsonl. Mirrors the Phase 1 TS decoder pattern in
fetch/src/decoders/uniswap-v3-swap.ts.
"""
from typing import Any
import json as _json
from pathlib import Path
import polars as pl
from eth_abi import decode as abi_decode

# --- Topic0 registry (verified against Phase 1 + canonical Uniswap V3 sources) ---
SWAP_TOPIC0 = "0xc42079f94a6350d7e6235f29174924f928cc2ac818eb64fed8004e115fbcca67"
MINT_TOPIC0 = "0x7a53080ba414158be7ec69b987b5fb7d07dee101fe85488f0853ae16239d0bde"
BURN_TOPIC0 = "0x0c396cd989a39f4459b5fa1aed6a9a8dcdbc45908acfd67e028cd568da98982c"
# ICHI vault Deposit/Withdraw — topic0 computed at module-load time from the captured ABI
# (analysis/tests/fixtures/ichi_vault_abi.json, produced by Plan 02-00 Task 3).
# This avoids paste-and-hope drift across ICHI vault versions: the canonical event
# signatures live in the ABI fixture; topic0 is derived via keccak256 of that signature.
def _compute_topic0_from_abi(abi: list, event_name: str) -> str:
    """Find an event entry by name in the captured ABI and return keccak256(canonical_signature)."""
    from eth_utils import keccak
    for entry in abi:
        if entry.get("type") == "event" and entry.get("name") == event_name:
            inputs = ",".join(inp["type"] for inp in entry.get("inputs", []))
            sig = f"{event_name}({inputs})".encode()
            return "0x" + keccak(sig).hex()
    raise KeyError(f"event {event_name!r} not found in captured ABI")

# Load the captured ICHI vault ABI once at module-load and compute topic0s
_ABI_FIXTURE_PATH = Path(__file__).resolve().parents[2] / "analysis" / "tests" / "fixtures" / "ichi_vault_abi.json"
try:
    with open(_ABI_FIXTURE_PATH, "r") as _f:
        ICHI_VAULT_ABI = _json.load(_f)
        # Tolerate captured ABI being a list, or wrapped under {"abi": [...]} or {"_meta": ..., "abi": [...]}
        if isinstance(ICHI_VAULT_ABI, dict) and "abi" in ICHI_VAULT_ABI:
            ICHI_VAULT_ABI = ICHI_VAULT_ABI["abi"]
    DEPOSIT_TOPIC0 = _compute_topic0_from_abi(ICHI_VAULT_ABI, "Deposit")
    WITHDRAW_TOPIC0 = _compute_topic0_from_abi(ICHI_VAULT_ABI, "Withdraw")
except (FileNotFoundError, KeyError) as _e:
    # Fallback for execution environments where the ABI fixture is absent (e.g., partial checkout).
    # The fallback values match common ICHI-V1 signatures `Deposit(address,address,uint256,uint256,uint256)`
    # and `Withdraw(address,address,uint256,uint256,uint256)`. Plan 02-09's acceptance grid
    # MUST verify the live values match by running the keccak check below.
    DEPOSIT_TOPIC0 = "0xdcbc1c05240f31ff3ad067ef1ee35ce4997762752e3a095284754544f4c709d7"
    WITHDRAW_TOPIC0 = "0xf279e6a1f5e320cca91135676d9cb6e44ca8a08c0b88342bcdb1144f6511b568"

TOPIC0_TO_EVENT = {
    SWAP_TOPIC0: "Swap",
    MINT_TOPIC0: "Mint",
    BURN_TOPIC0: "Burn",
    DEPOSIT_TOPIC0: "Deposit",
    WITHDRAW_TOPIC0: "Withdraw",
}

def _hex_to_bytes(h: str) -> bytes:
    return bytes.fromhex(h[2:] if h.startswith("0x") else h)

def _topic_to_int(t: str, signed: bool = False, bits: int = 256) -> int:
    v = int(t, 16)
    if signed and v >= (1 << (bits - 1)):
        v -= 1 << bits
    return v

def _topic_to_addr(t: str) -> str:
    # Last 20 bytes of a 32-byte topic
    return "0x" + t[-40:].lower()

def decode_swap(topics: list[str], data: str) -> dict[str, Any]:
    sender = _topic_to_addr(topics[1])
    recipient = _topic_to_addr(topics[2])
    (amount0, amount1, sqrtPriceX96, liquidity, tick) = abi_decode(
        ["int256", "int256", "uint160", "uint128", "int24"],
        _hex_to_bytes(data),
    )
    return {
        "sender": sender, "recipient": recipient,
        "amount0": str(amount0), "amount1": str(amount1),
        "sqrtPriceX96": str(sqrtPriceX96),
        "liquidity": str(liquidity),
        "tick": int(tick),
    }

def decode_mint(topics: list[str], data: str) -> dict[str, Any]:
    owner = _topic_to_addr(topics[1])
    tickLower = _topic_to_int(topics[2], signed=True, bits=24)
    tickUpper = _topic_to_int(topics[3], signed=True, bits=24)
    (sender, amount, amount0, amount1) = abi_decode(
        ["address", "uint128", "uint256", "uint256"],
        _hex_to_bytes(data),
    )
    return {
        "sender": sender, "owner": owner,
        "tickLower": int(tickLower), "tickUpper": int(tickUpper),
        "amount": str(amount), "amount0": str(amount0), "amount1": str(amount1),
    }

def decode_burn(topics: list[str], data: str) -> dict[str, Any]:
    owner = _topic_to_addr(topics[1])
    tickLower = _topic_to_int(topics[2], signed=True, bits=24)
    tickUpper = _topic_to_int(topics[3], signed=True, bits=24)
    (amount, amount0, amount1) = abi_decode(
        ["uint128", "uint256", "uint256"],
        _hex_to_bytes(data),
    )
    return {
        "owner": owner,
        "tickLower": int(tickLower), "tickUpper": int(tickUpper),
        "amount": str(amount), "amount0": str(amount0), "amount1": str(amount1),
    }

def decode_deposit(topics: list[str], data: str) -> dict[str, Any]:
    sender = _topic_to_addr(topics[1])
    to = _topic_to_addr(topics[2])
    (shares, amount0, amount1) = abi_decode(
        ["uint256", "uint256", "uint256"],
        _hex_to_bytes(data),
    )
    return {"sender": sender, "to": to, "shares": str(shares),
            "amount0": str(amount0), "amount1": str(amount1)}

def decode_withdraw(topics: list[str], data: str) -> dict[str, Any]:
    sender = _topic_to_addr(topics[1])
    to = _topic_to_addr(topics[2])
    (shares, amount0, amount1) = abi_decode(
        ["uint256", "uint256", "uint256"],
        _hex_to_bytes(data),
    )
    return {"sender": sender, "to": to, "shares": str(shares),
            "amount0": str(amount0), "amount1": str(amount1)}

_DECODER_FUNCS = {
    "Swap": decode_swap,
    "Mint": decode_mint,
    "Burn": decode_burn,
    "Deposit": decode_deposit,
    "Withdraw": decode_withdraw,
}

def decode_all(df: pl.DataFrame) -> pl.DataFrame:
    """Decode every row's raw `topics` + `data` into typed payload columns per event_name.

    Adds `event_name` column from topics[0] via TOPIC0_TO_EVENT lookup
    (unknown topic0 → event_name='Unknown'; row preserved).

    For each known event_name, decodes the payload via the corresponding decode_* function
    and adds the resulting fields as columns. Numeric fields are emitted as decimal-strings
    (polars String dtype) to preserve uint256 precision; Plan 02-05 casts to Decimal[38,18]
    as needed for arithmetic.
    """
    rows = df.to_dicts()
    enriched = []
    for r in rows:
        topics = r.get("topics") or []
        topic0 = topics[0] if topics else ""
        event_name = TOPIC0_TO_EVENT.get(topic0, "Unknown")
        out = dict(r)
        out["event_name"] = event_name
        decoder = _DECODER_FUNCS.get(event_name)
        if decoder is not None:
            try:
                out.update(decoder(topics, r.get("data", "0x")))
            except Exception as e:
                # Surface decode failures explicitly per row but don't raise — Plan 02-08 audits
                out["decode_error"] = str(e)
        enriched.append(out)
    return pl.DataFrame(enriched)
```
